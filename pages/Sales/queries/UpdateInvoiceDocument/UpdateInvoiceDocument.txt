UPDATE documents
SET
    document_type = {{ this.params.documentType || "SALES_INVOICE" }},
    document_number = {{ this.params.documentNumber }},
    document_date = {{ this.params.documentDate }},
    due_date = {{ this.params.dueDate || this.params.documentDate }},
    partner_id = {{ Number(this.params.customerId || 0) }},
    warehouse_id = {{ Number(this.params.warehouseId || 0) }},
    payment_method = {{ this.params.paymentMethod || null }},
    currency_code = {{ this.params.currencyCode || "EUR" }},
    exchange_rate = {{ Number(this.params.exchangeRate || 1) }},
    subtotal_amount = {{ Number(this.params.subtotalAmount || 0) }},
    discount_amount = {{ Number(this.params.discountAmount || 0) }},
    tax_amount = {{ Number(this.params.taxAmount || 0) }},
    total_amount = {{ Number(this.params.totalAmount || 0) }},
    note = {{ this.params.note || null }},
    updated_at = NOW()
WHERE id = {{ Number(this.params.documentId || 0) }}
  AND company_id = {{ Number(appsmith.store.companyId || 1) }}
  AND document_type IN ('SALES_INVOICE', 'CREDIT_NOTE')
  AND status = 'DRAFT';
