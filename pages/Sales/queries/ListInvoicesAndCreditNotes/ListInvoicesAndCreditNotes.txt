SELECT
    d.id AS documentId,
    d.document_number AS `Document Number`,
    COALESCE(d.fiscal_number, d.document_number) AS `Fiscal Number`,
    d.document_type AS documentType,

    CASE
      WHEN d.document_type = 'CREDIT_NOTE' THEN 'Credit Note'
      ELSE 'Invoice'
    END AS `Type`,

    d.document_date AS `Document Date`,
    DATE_FORMAT(d.created_at, '%d.%m.%Y %H:%i:%s') AS `Issue Time`,

    bp.code AS `Customer Code`,
    bp.name AS `Customer`,

    d.payment_method AS `Payment Method`,
    d.currency_code AS `Currency`,
    COALESCE(d.exchange_rate, 1) AS `Exchange Rate`,

    CASE
      WHEN d.document_type = 'CREDIT_NOTE' THEN -ABS(COALESCE(d.subtotal_amount, 0))
      ELSE COALESCE(d.subtotal_amount, 0)
    END AS `Subtotal`,

    CASE
      WHEN d.document_type = 'CREDIT_NOTE' THEN -ABS(COALESCE(d.discount_amount, 0))
      ELSE COALESCE(d.discount_amount, 0)
    END AS `Discount`,

    CASE
      WHEN d.document_type = 'CREDIT_NOTE' THEN -ABS(COALESCE(d.tax_amount, 0))
      ELSE COALESCE(d.tax_amount, 0)
    END AS `Tax`,

    CASE
      WHEN d.document_type = 'CREDIT_NOTE' THEN -ABS(COALESCE(d.total_amount, 0))
      ELSE COALESCE(d.total_amount, 0)
    END AS `Total`,

    CASE
      WHEN d.document_type = 'CREDIT_NOTE' THEN -ABS(COALESCE(d.paid_amount, 0))
      ELSE COALESCE(d.paid_amount, 0)
    END AS `Paid`,

    CASE
      WHEN d.document_type = 'CREDIT_NOTE'
      THEN -ABS(COALESCE(d.total_amount, 0) - COALESCE(d.paid_amount, 0))
      ELSE COALESCE(d.total_amount, 0) - COALESCE(d.paid_amount, 0)
    END AS `Open Amount`,

    d.status AS `Status`,
    d.posting_status AS `Posting Status`,
    d.fiscal_status AS `Fiscal Status`,

    COALESCE(d.fiscal_jir, d.jir) AS jir,
    COALESCE(d.fiscal_zki, d.zki) AS zki,

    w.code AS `Warehouse Code`,
    w.name AS `Warehouse`,

    u.username AS `Created By`,
    d.created_at AS createdAt,
    d.updated_at AS updatedAt

FROM documents d
LEFT JOIN business_partners bp ON bp.id = d.partner_id
LEFT JOIN warehouses w ON w.id = d.warehouse_id
LEFT JOIN users u ON u.id = d.created_by_user_id

WHERE d.company_id = {{ Number(appsmith.store.companyId || 1) }}
  AND d.document_type IN ('SALES_INVOICE', 'CREDIT_NOTE')

  AND (
      {{ InvoiceSearchInput.text || "" }} = ""
      OR d.document_number LIKE CONCAT('%', {{ InvoiceSearchInput.text || "" }}, '%')
      OR COALESCE(d.fiscal_number, '') LIKE CONCAT('%', {{ InvoiceSearchInput.text || "" }}, '%')
      OR COALESCE(bp.name, '') LIKE CONCAT('%', {{ InvoiceSearchInput.text || "" }}, '%')
      OR COALESCE(bp.code, '') LIKE CONCAT('%', {{ InvoiceSearchInput.text || "" }}, '%')
  )

  AND (
      {{ InvoiceTypeSelect.selectedOptionValue || "ALL" }} = "ALL"
      OR d.document_type = {{ InvoiceTypeSelect.selectedOptionValue || "ALL" }}
  )

  AND (
      {{ InvoiceStatusSelect.selectedOptionValue || "ALL" }} = "ALL"
      OR d.status = {{ InvoiceStatusSelect.selectedOptionValue || "ALL" }}
  )

  AND (
      {{ Number(InvoiceCustomerSelect.selectedOptionValue || 0) }} = 0
      OR d.partner_id = {{ Number(InvoiceCustomerSelect.selectedOptionValue || 0) }}
  )

  AND (
      {{ Number(InvoiceWarehouseSelect.selectedOptionValue || 0) }} = 0
      OR d.warehouse_id = {{ Number(InvoiceWarehouseSelect.selectedOptionValue || 0) }}
  )

  AND d.document_date >= {{ moment(InvoiceDateFrom.selectedDate || moment().startOf("year")).format("YYYY-MM-DD") }}
  AND d.document_date <= {{ moment(InvoiceDateTo.selectedDate || moment()).format("YYYY-MM-DD") }}

ORDER BY d.document_date DESC, d.id DESC;
