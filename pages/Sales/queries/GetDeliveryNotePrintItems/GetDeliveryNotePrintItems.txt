SELECT
    di.id AS documentItemId,
    di.line_no AS lineNo,
    di.product_id AS productId,

    p.code AS productCode,
    p.name AS productName,
    p.barcode AS barcode,
    p.sku AS sku,

    COALESCE(di.description, p.name) AS description,

    di.quantity AS quantity,
    u.code AS unitCode,

    di.batch_number AS batchNumber,
    di.serial_number AS serialNumber,
    di.expiry_date AS expiryDate,

    di.note AS note

FROM document_items di

LEFT JOIN products p
  ON p.id = di.product_id

LEFT JOIN units_of_measure u
  ON u.id = di.unit_id

WHERE di.document_id = {{ Number(this.params.documentId || appsmith.store.currentDeliveryNotePrintId || 0) }}

ORDER BY di.line_no;
