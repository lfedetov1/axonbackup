INSERT INTO document_items (
    document_id,
    line_no,
    product_id,
    description,
    quantity,
    unit_id,
    unit_price,
    discount_amount,
    line_subtotal,
    tax_rate_id,
    tax_amount,
    line_total,
    note
)
SELECT
    {{this.params.invoiceId}},
    line_no,
    product_id,
    description,
    quantity,
    unit_id,
    unit_price,
    discount_amount,
    line_subtotal,
    tax_rate_id,
    tax_amount,
    line_total,
    CONCAT('From sales order item ', id)
FROM document_items
WHERE document_id = {{this.params.documentId}}
ORDER BY line_no;
