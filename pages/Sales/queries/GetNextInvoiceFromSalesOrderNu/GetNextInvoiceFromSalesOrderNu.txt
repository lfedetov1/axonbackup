SELECT
    CONCAT(
      'INV-',
      DATE_FORMAT(CURDATE(), '%Y%m%d'),
      '-',
      LPAD(
        COALESCE(MAX(CAST(SUBSTRING_INDEX(document_number, '-', -1) AS UNSIGNED)), 0) + 1,
        6,
        '0'
      )
    ) AS nextInvoiceNumber
FROM documents
WHERE company_id = {{appsmith.store.companyId}}
  AND document_type = 'SALES_INVOICE'
  AND document_number LIKE CONCAT('INV-', DATE_FORMAT(CURDATE(), '%Y%m%d'), '-%');
