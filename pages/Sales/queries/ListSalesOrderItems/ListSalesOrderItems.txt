SELECT
    di.id AS documentItemId,
    di.line_no AS `Line`,
    p.code AS `Product Code`,
    p.name AS `Product`,
    p.barcode AS `Barcode`,
    di.description AS `Description`,
    di.quantity AS `Quantity`,
    u.code AS `Unit`,
    di.unit_price AS `Unit Price`,
    di.discount_amount AS `Discount`,
    tr.rate AS `Tax Rate`,
    di.tax_amount AS `Tax`,
    di.line_total AS `Total`,
    di.note AS `Note`

FROM document_items di
LEFT JOIN products p ON p.id = di.product_id
LEFT JOIN units_of_measure u ON u.id = di.unit_id
LEFT JOIN tax_rates tr ON tr.id = di.tax_rate_id

WHERE di.document_id = {{
  SalesOrdersTable.selectedRow.documentId ||
  SalesOrdersTable.selectedRow.id ||
  SalesOrdersTable.selectedRow.ID ||
  0
}}

ORDER BY di.line_no;
