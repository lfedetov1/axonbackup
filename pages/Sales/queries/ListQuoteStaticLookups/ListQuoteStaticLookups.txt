SELECT 'PAYMENT_METHOD' AS lookupType, 'Bank Transfer' AS label, 'BANK_TRANSFER' AS value, 10 AS sortOrder
UNION ALL SELECT 'PAYMENT_METHOD', 'Cash', 'CASH', 20
UNION ALL SELECT 'PAYMENT_METHOD', 'Card', 'CARD', 30
UNION ALL SELECT 'PAYMENT_METHOD', 'Other', 'OTHER', 90

UNION ALL SELECT 'CURRENCY', 'EUR', 'EUR', 10
UNION ALL SELECT 'CURRENCY', 'USD', 'USD', 20
UNION ALL SELECT 'CURRENCY', 'BAM', 'BAM', 30

UNION ALL SELECT 'SALES_CHANNEL', 'Direct', 'DIRECT', 10
UNION ALL SELECT 'SALES_CHANNEL', 'Web', 'WEB', 20
UNION ALL SELECT 'SALES_CHANNEL', 'Retail', 'RETAIL', 30
UNION ALL SELECT 'SALES_CHANNEL', 'Phone / Email', 'PHONE_EMAIL', 40

ORDER BY lookupType, sortOrder;
