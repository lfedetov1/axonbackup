SELECT
    d.id AS documentId,
    d.document_number AS documentNumber,
    d.document_date AS documentDate,
    d.valid_until AS validUntil,
    d.status AS status,
    d.partner_id AS partnerId,
    d.warehouse_id AS warehouseId,
    d.currency_code AS currencyCode,
    d.sales_channel AS salesChannel,
    d.note AS note
FROM documents d
WHERE d.id = {{ Number(this.params.documentId || 0) }}
  AND d.document_type = 'QUOTE'
LIMIT 1;
