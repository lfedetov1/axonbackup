SELECT
    u.id AS userId,
    u.username,

    GROUP_CONCAT(DISTINCT r.code ORDER BY r.code SEPARATOR ',') AS roleCodes,
    GROUP_CONCAT(DISTINCT p.code ORDER BY p.code SEPARATOR ',') AS permissionCodes,
    GROUP_CONCAT(DISTINCT uwa.warehouse_id ORDER BY uwa.warehouse_id SEPARATOR ',') AS warehouseIds,

    MAX(CASE WHEN r.code = 'ADMIN' THEN 1 ELSE 0 END) AS isAdmin,
    MAX(
      CASE
        WHEN r.code = 'ADMIN' THEN 1
        WHEN p.code = 'inventory.warehouse_all' THEN 1
        WHEN p.code = 'sales.view_all_warehouses' THEN 1
        ELSE 0
      END
    ) AS canViewAllWarehouses

FROM users u

LEFT JOIN user_role_assignments ura
  ON ura.user_id = u.id

LEFT JOIN roles r
  ON r.id = ura.role_id

LEFT JOIN role_permissions rp
  ON rp.role_id = r.id

LEFT JOIN permissions p
  ON p.id = rp.permission_id

LEFT JOIN user_warehouse_access uwa
  ON uwa.user_id = u.id

WHERE u.id = {{ Number(appsmith.store.userId || 0) }}

GROUP BY
    u.id,
    u.username;
