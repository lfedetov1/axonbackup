SELECT
    d.id AS documentId,
    d.document_number AS documentNumber,
    d.document_date AS documentDate,
    d.status AS status,
    d.posting_status AS postingStatus,
    d.source_document_id AS salesOrderId,
    d.created_at AS createdAt
FROM documents d
WHERE d.document_type = 'DELIVERY_NOTE'
  AND d.source_document_id = {{ Number(appsmith.store.currentSalesOrderShippingId || SalesOrderTable.selectedRow.documentId || 0) }}
ORDER BY d.id DESC;
