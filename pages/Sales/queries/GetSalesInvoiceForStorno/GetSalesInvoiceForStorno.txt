SELECT
    d.id AS invoiceId,
    d.document_number AS invoiceNumber,
    d.document_date AS invoiceDate,
    d.company_id AS companyId,
    d.partner_id AS partnerId,
    d.warehouse_id AS warehouseId,
    d.currency_code AS currencyCode,
    d.exchange_rate AS exchangeRate,
    d.payment_method AS paymentMethod,
    d.subtotal_amount AS subtotalAmount,
    d.discount_amount AS discountAmount,
    d.tax_amount AS taxAmount,
    d.total_amount AS totalAmount,
    d.status,
    d.posting_status AS postingStatus,
    d.fiscal_status AS fiscalStatus
FROM documents d
WHERE d.id = {{ Number((this.params && this.params.invoiceId) || appsmith.store.currentInvoiceId || 0) }}
  AND d.company_id = {{ Number(appsmith.store.companyId || 1) }}
  AND d.document_type = 'SALES_INVOICE'
  AND d.status <> 'CANCELLED'
LIMIT 1;
