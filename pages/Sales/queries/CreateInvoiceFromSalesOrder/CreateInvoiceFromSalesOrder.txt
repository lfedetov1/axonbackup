INSERT INTO documents (
    company_id,
    warehouse_id,
    partner_id,
    created_by_user_id,
    document_type,
    status,
    document_number,
    document_date,
    due_date,
    currency_code,
    source_document_id,
    sales_channel,
    note,
    subtotal_amount,
    tax_amount,
    discount_amount,
    total_amount,
    paid_amount
)
SELECT
    company_id,
    warehouse_id,
    partner_id,
    {{appsmith.store.userId}},
    'SALES_INVOICE',
    'DRAFT',
    {{this.params.invoiceNumber}},
    CURDATE(),
    due_date,
    currency_code,
    id,
    sales_channel,
    CONCAT('Invoice from sales order ', document_number),
    subtotal_amount,
    tax_amount,
    discount_amount,
    total_amount,
    0
FROM documents
WHERE id = {{this.params.documentId}}
  AND document_type = 'SALES_ORDER';
