SELECT
    dp.id AS packageId,
    dp.package_number AS packageNumber,
    dp.package_barcode AS packageBarcode,
    dp.package_type AS packageType,
    dp.package_status AS packageStatus,

    dp.gross_weight AS grossWeight,
    dp.net_weight AS netWeight,
    dp.length_cm AS lengthCm,
    dp.width_cm AS widthCm,
    dp.height_cm AS heightCm,
    dp.item_count AS itemCount,
    dp.total_quantity AS totalQuantity,
    dp.note AS packageNote,

    d.id AS documentId,
    d.document_number AS deliveryNumber,
    d.document_date AS deliveryDate,

    so.document_number AS salesOrderNumber,

    c.name AS companyName,
    c.tax_number AS companyTaxNumber,
    c.email AS companyEmail,
    c.phone AS companyPhone,
    c.address_line1 AS companyAddress,
    c.city AS companyCity,
    c.postal_code AS companyPostalCode,
    c.logo_path AS companyLogoPath,

    w.code AS warehouseCode,
    w.name AS warehouseName,

    bp.code AS customerCode,
    COALESCE(bp.legal_name, bp.name) AS customerName,
    bp.tax_number AS customerTaxNumber,
    bp.email AS customerEmail,
    bp.phone AS customerPhone,
    bp.address_line1 AS customerAddress,
    bp.city_name AS customerCity,
    bp.postal_code AS customerPostalCode

FROM document_packages dp
JOIN documents d ON d.id = dp.document_id
JOIN companies c ON c.id = dp.company_id
LEFT JOIN documents so ON so.id = d.source_document_id
LEFT JOIN warehouses w ON w.id = dp.warehouse_id
LEFT JOIN business_partners bp ON bp.id = dp.partner_id

WHERE dp.id = {{ Number(appsmith.store.selectedPackagePrintId || 0) }}
LIMIT 1;
