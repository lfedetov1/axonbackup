SELECT * FROM accountaSELECT
    p.id AS productId,
    p.code AS productCode,
    p.name AS productName,
    p.barcode,
    p.sku,
    p.product_type AS productType,
    p.track_stock AS trackStock,

    u.id AS unitId,
    u.code AS unitCode,

    COALESCE(pp.price, 0) AS unitPrice,

    tr.id AS taxRateId,
    tr.rate AS taxRate,

    COALESCE(stock.currentStock, 0) AS availableStock

FROM products p
LEFT JOIN units_of_measure u ON u.id = p.unit_id
LEFT JOIN tax_rates tr ON tr.id = p.tax_rate_id

LEFT JOIN (
    SELECT
        product_id,
        MAX(price) AS price
    FROM product_prices
    GROUP BY product_id
) pp ON pp.product_id = p.id

LEFT JOIN (
    SELECT
        product_id,
        SUM(
            CASE
              WHEN movement_type IN ('IN', 'TRANSFER_IN', 'ADJUSTMENT') THEN quantity
              WHEN movement_type IN ('OUT', 'TRANSFER_OUT') THEN -quantity
              ELSE 0
            END
        ) AS currentStock
    FROM stock_movements
    WHERE warehouse_id = {{this.params.warehouseId || SalesOrderWarehouseSelect.selectedOptionValue || appsmith.store.warehouseId || 0}}
    GROUP BY product_id
) stock ON stock.product_id = p.id

WHERE p.is_active = 1
  AND (
      p.barcode = {{this.params.lookup}}
      OR p.code = {{this.params.lookup}}
      OR p.sku = {{this.params.lookup}}
      OR p.name LIKE CONCAT('%', {{this.params.lookup}}, '%')
  )
ORDER BY
    CASE
      WHEN p.barcode = {{this.params.lookup}} THEN 1
      WHEN p.code = {{this.params.lookup}} THEN 2
      WHEN p.sku = {{this.params.lookup}} THEN 3
      ELSE 4
    END
LIMIT 10;ing_accounts LIMIT 10;
