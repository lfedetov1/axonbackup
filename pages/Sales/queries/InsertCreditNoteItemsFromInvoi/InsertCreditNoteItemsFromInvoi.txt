INSERT INTO document_items (
    document_id, line_no, product_id, description, quantity, unit_id,
    unit_price, discount_amount, tax_rate_id, tax_amount,
    line_subtotal, line_total, note
)
SELECT
    {{ Number((this.params && this.params.creditNoteId) || 0) }},
    di.line_no,
    di.product_id,
    di.description,
    ABS(di.quantity),
    di.unit_id,
    di.unit_price,
    COALESCE(di.discount_amount, 0),
    di.tax_rate_id,
    COALESCE(di.tax_amount, 0),
    COALESCE(di.line_subtotal, di.quantity * di.unit_price, 0),
    COALESCE(di.line_total, 0),
    CONCAT('Storno item from invoice item ', di.id)
FROM document_items di
JOIN documents inv ON inv.id = di.document_id
WHERE inv.id = {{ Number((this.params && this.params.invoiceId) || 0) }}
  AND inv.company_id = {{ Number(appsmith.store.companyId || 1) }}
  AND inv.document_type = 'SALES_INVOICE';
