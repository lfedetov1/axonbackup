SELECT
    d.id AS documentId,
    d.document_number AS `Quote Number`,
    d.document_date AS `Quote Date`,
    DATE_FORMAT(d.created_at, '%d.%m.%Y %H:%i:%s') AS `Issue Time`,

    d.valid_until AS `Valid Until`,
    d.due_date AS `Due Date`,

    bp.code AS `Customer Code`,
    bp.name AS `Customer`,

    d.payment_method AS `Payment Method`,
    d.currency_code AS `Currency`,
    d.exchange_rate AS `Exchange Rate`,
    d.sales_channel AS `Sales Channel`,

    d.subtotal_amount AS `Subtotal`,
    d.discount_amount AS `Discount`,
    d.tax_amount AS `Tax`,
    d.total_amount AS `Total`,

    d.status AS `Status`,
    d.posting_status AS `Posting Status`,

    w.code AS `Warehouse Code`,
    w.name AS `Warehouse`,

    u.username AS `Created By`,
    d.created_at AS createdAt,
    d.updated_at AS updatedAt

FROM documents d
LEFT JOIN business_partners bp
  ON bp.id = d.partner_id

LEFT JOIN warehouses w
  ON w.id = d.warehouse_id

LEFT JOIN users u
  ON u.id = d.created_by_user_id

WHERE d.company_id = {{ Number(appsmith.store.companyId || 0) }}
  AND d.document_type = 'QUOTE'

  AND (
      {{ QuoteSearchInput.text || "" }} = ""
      OR d.document_number LIKE CONCAT('%', {{ QuoteSearchInput.text || "" }}, '%')
      OR bp.name LIKE CONCAT('%', {{ QuoteSearchInput.text || "" }}, '%')
      OR bp.code LIKE CONCAT('%', {{ QuoteSearchInput.text || "" }}, '%')
  )

  AND (
      {{ Number(QuoteCustomerSelect.selectedOptionValue || 0) }} = 0
      OR d.partner_id = {{ Number(QuoteCustomerSelect.selectedOptionValue || 0) }}
  )

  AND (
      {{ Number(QuoteWarehouseSelect.selectedOptionValue || 0) }} = 0
      OR d.warehouse_id = {{ Number(QuoteWarehouseSelect.selectedOptionValue || 0) }}
  )

  AND d.document_date >= {{ moment(QuoteDateFrom.selectedDate || moment().startOf("month")).format("YYYY-MM-DD") }}
  AND d.document_date <= {{ moment(QuoteDateTo.selectedDate || moment()).format("YYYY-MM-DD") }}

ORDER BY d.document_date DESC, d.id DESC;
