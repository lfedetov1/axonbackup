SELECT
    w.id AS warehouseId,
    w.code AS warehouseCode,
    w.name AS warehouseName,
    p.id AS productId,
    p.code AS productCode,
    p.name AS productName,
    COALESCE(stock.currentStock, 0) AS availableStock
FROM document_items di
JOIN products p ON p.id = di.product_id
JOIN warehouses w ON w.company_id = {{appsmith.store.companyId}}
LEFT JOIN (
    SELECT
        warehouse_id,
        product_id,
        SUM(
            CASE
              WHEN movement_type IN ('IN', 'TRANSFER_IN') THEN quantity
              WHEN movement_type IN ('OUT', 'TRANSFER_OUT') THEN -quantity
              WHEN movement_type = 'ADJUSTMENT' THEN quantity
              ELSE 0
            END
        ) AS currentStock
    FROM stock_movements
    GROUP BY warehouse_id, product_id
) stock ON stock.warehouse_id = w.id AND stock.product_id = di.product_id
WHERE di.document_id = {{this.params.documentId}}
  AND w.id <> {{this.params.destinationWarehouseId}}
  AND COALESCE(stock.currentStock, 0) >= di.quantity
ORDER BY availableStock DESC;
