SELECT
    d.id AS invoiceId,
    d.id AS documentId,
    d.document_number AS invoiceNumber,
    d.document_number AS documentNumber
FROM documents d
WHERE d.company_id = {{ Number(appsmith.store.companyId || 0) }}
  AND d.document_type = 'SALES_INVOICE'
  AND (
      d.document_number = {{ this.params.invoiceNumber || "" }}
      OR d.source_document_id = {{ Number(this.params.quoteId || this.params.documentId || 0) }}
  )
ORDER BY d.id DESC
LIMIT 1;
