SELECT
    bp.id AS customerId,
    bp.code AS customerCode,
    bp.name AS customerName,
    COUNT(*) AS documentCount,
    COALESCE(SUM(
      CASE
        WHEN d.document_type = 'CREDIT_NOTE' THEN -ABS(COALESCE(d.total_amount, 0))
        ELSE COALESCE(d.total_amount, 0)
      END
    ), 0) AS totalAmount,
    COALESCE(SUM(
      CASE
        WHEN d.document_type = 'CREDIT_NOTE'
        THEN -ABS(COALESCE(d.total_amount, 0) - COALESCE(d.paid_amount, 0))
        ELSE COALESCE(d.total_amount, 0) - COALESCE(d.paid_amount, 0)
      END
    ), 0) AS openAmount
FROM documents d
LEFT JOIN business_partners bp ON bp.id = d.partner_id
WHERE d.company_id = {{ Number(appsmith.store.companyId || 1) }}
  AND d.document_type IN ('SALES_INVOICE', 'CREDIT_NOTE')
  AND d.status <> 'CANCELLED'
  AND d.document_date >= {{ moment(SalesAnalysisDateFrom.selectedDate || moment().startOf("month")).format("YYYY-MM-DD") }}
  AND d.document_date <= {{ moment(SalesAnalysisDateTo.selectedDate || moment()).format("YYYY-MM-DD") }}
  AND (
      {{ Number(SalesAnalysisWarehouseSelect.selectedOptionValue || appsmith.store.warehouseId1 || appsmith.store.warehouseId || 0) }} = 0
      OR d.warehouse_id = {{ Number(SalesAnalysisWarehouseSelect.selectedOptionValue || appsmith.store.warehouseId1 || appsmith.store.warehouseId || 0) }}
  )
GROUP BY bp.id, bp.code, bp.name
ORDER BY totalAmount DESC
LIMIT 10;
