SELECT
    d.id AS salesOrderId,
    d.company_id AS companyId,
    d.document_number AS salesOrderNumber,
    d.document_date AS salesOrderDate,
    d.partner_id AS partnerId,
    d.warehouse_id AS warehouseId,
    d.currency_code AS currencyCode,
    d.payment_method AS paymentMethod,
    d.sales_channel AS salesChannel,
    d.fulfillment_type AS fulfillmentType,
		d.subtotal_amount AS subtotalAmount,
    d.discount_amount AS discountAmount,
    d.tax_amount AS taxAmount,
    d.total_amount AS totalAmount,
    d.status AS status,
    d.note AS note,
    d.created_by_user_id AS createdByUserId
FROM documents d
WHERE d.id = {{ Number(this.params.salesOrderId || 0) }}
  AND d.document_type = 'SALES_ORDER'
LIMIT 1;
