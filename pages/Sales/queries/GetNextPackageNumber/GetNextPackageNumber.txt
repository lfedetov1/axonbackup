SELECT
  CONCAT(
    'PKG-',
    DATE_FORMAT(CURDATE(), '%Y%m%d'),
    '-',
    LPAD(
      COALESCE(
        MAX(CAST(SUBSTRING_INDEX(package_number, '-', -1) AS UNSIGNED)),
        0
      ) + 1,
      4,
      '0'
    )
  ) AS nextPackageNumber
FROM document_packages
WHERE company_id = {{ Number(appsmith.store.companyId || 0) }}
  AND package_number LIKE CONCAT('PKG-', DATE_FORMAT(CURDATE(), '%Y%m%d'), '-%');
