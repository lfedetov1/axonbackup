SELECT
    d.id AS documentId,
    d.document_number AS documentNumber,
    d.document_type AS documentType,

    CASE
      WHEN d.document_type = 'QUOTE' THEN 'Quote'
      WHEN d.document_type = 'SALES_ORDER' THEN 'Sales Order'
      WHEN d.document_type = 'DELIVERY_NOTE' THEN 'Delivery Note'
      WHEN d.document_type = 'CREDIT_NOTE' THEN 'Credit Note'
      WHEN d.document_type = 'SALES_INVOICE' THEN 'Sales Invoice'
      ELSE d.document_type
    END AS documentTitle,

    COALESCE(d.fiscal_number, d.document_number) AS fiscalNumber,
    COALESCE(d.fiscal_jir, d.jir) AS jir,
    COALESCE(d.fiscal_zki, d.zki) AS zki,
    COALESCE(d.fiscal_qr_url, d.fiscal_verification_url) AS fiscalVerificationUrl,
    d.fiscalized_at AS fiscalizedAt,
    d.fiscal_status AS fiscalStatus,

    d.document_date AS documentDate,
    d.valid_until AS validUntil,
    d.due_date AS dueDate,
    DATE_FORMAT(d.created_at, '%d.%m.%Y %H:%i:%s') AS issueTime,

    d.payment_method AS paymentMethod,
    d.currency_code AS currencyCode,
    d.exchange_rate AS exchangeRate,
    d.sales_channel AS salesChannel,

    CASE WHEN d.document_type = 'CREDIT_NOTE' THEN -ABS(COALESCE(d.subtotal_amount, 0)) ELSE COALESCE(d.subtotal_amount, 0) END AS subtotalAmount,
    CASE WHEN d.document_type = 'CREDIT_NOTE' THEN -ABS(COALESCE(d.discount_amount, 0)) ELSE COALESCE(d.discount_amount, 0) END AS discountAmount,
    CASE WHEN d.document_type = 'CREDIT_NOTE' THEN -ABS(COALESCE(d.tax_amount, 0)) ELSE COALESCE(d.tax_amount, 0) END AS taxAmount,
    CASE WHEN d.document_type = 'CREDIT_NOTE' THEN -ABS(COALESCE(d.total_amount, 0)) ELSE COALESCE(d.total_amount, 0) END AS totalAmount,
    CASE WHEN d.document_type = 'CREDIT_NOTE' THEN -ABS(COALESCE(d.paid_amount, 0)) ELSE COALESCE(d.paid_amount, 0) END AS paidAmount,

    d.status,
    d.posting_status AS postingStatus,
    d.note,

    bp.code AS partnerCode,
    bp.name AS partnerName,
    bp.legal_name AS partnerLegalName,
    bp.tax_number AS partnerTaxNumber,
    bp.registration_number AS partnerRegistrationNumber,
    bp.email AS partnerEmail,
    bp.phone AS partnerPhone,
    bp.address_line1 AS partnerAddress,
    bp.city_name AS partnerCity,
    bp.postal_code AS partnerPostalCode,

    c.name AS companyName,
    c.tax_number AS companyTaxNumber,
    c.registration_number AS companyRegistrationNumber,
    c.email AS companyEmail,
    c.phone AS companyPhone,
    c.address_line1 AS companyAddress,
    c.city AS companyCity,
    c.postal_code AS companyPostalCode,
    c.logo_path AS companyLogoPath,

    w.code AS warehouseCode,
    w.name AS warehouseName,

    u.username AS createdBy
FROM documents d
JOIN companies c ON c.id = d.company_id
LEFT JOIN business_partners bp ON bp.id = d.partner_id
LEFT JOIN warehouses w ON w.id = d.warehouse_id
LEFT JOIN users u ON u.id = d.created_by_user_id
WHERE d.id = {{ Number((this.params && this.params.documentId) || appsmith.store.salesSelectedDocumentId || 0) }}
  AND d.company_id = {{ Number(appsmith.store.companyId || 1) }}
  AND d.document_type IN ('QUOTE', 'SALES_ORDER', 'DELIVERY_NOTE', 'SALES_INVOICE', 'CREDIT_NOTE')
LIMIT 1;
