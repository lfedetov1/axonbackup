SELECT
    di.id AS documentItemId,
    di.product_id AS productId,
    p.code AS productCode,
    p.name AS productName,
    di.quantity AS requiredQuantity,
    COALESCE(stock.currentStock, 0) AS availableStock,
    GREATEST(di.quantity - COALESCE(stock.currentStock, 0), 0) AS shortageQuantity
FROM document_items di
JOIN documents d ON d.id = di.document_id
JOIN products p ON p.id = di.product_id
LEFT JOIN (
    SELECT
        product_id,
        SUM(
            CASE
              WHEN movement_type IN ('IN', 'TRANSFER_IN') THEN quantity
              WHEN movement_type IN ('OUT', 'TRANSFER_OUT') THEN -quantity
              WHEN movement_type = 'ADJUSTMENT' THEN quantity
              ELSE 0
            END
        ) AS currentStock
    FROM stock_movements
    WHERE warehouse_id = {{this.params.warehouseId}}
    GROUP BY product_id
) stock ON stock.product_id = di.product_id
WHERE di.document_id = {{this.params.documentId}};
