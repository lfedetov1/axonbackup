SELECT
    CONCAT(
      'SO-',
      DATE_FORMAT(CURDATE(), '%Y%m%d'),
      '-',
      LPAD(
        COALESCE(
          MAX(CAST(SUBSTRING_INDEX(document_number, '-', -1) AS UNSIGNED)),
          0
        ) + 1,
        6,
        '0'
      )
    ) AS nextSalesOrderNumber
FROM documents
WHERE company_id = {{appsmith.store.companyId}}
  AND document_type = 'SALES_ORDER'
  AND document_number LIKE CONCAT('SO-', DATE_FORMAT(CURDATE(), '%Y%m%d'), '-%');
