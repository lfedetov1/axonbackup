SELECT
    d.id AS deliveryNoteId,
    d.company_id AS companyId,
    d.source_document_id AS salesOrderId,
    d.partner_id AS partnerId,
    d.warehouse_id AS warehouseId,
    d.document_number AS deliveryNoteNumber,
    d.created_by_user_id AS createdByUserId
FROM documents d
WHERE d.document_type = 'DELIVERY_NOTE'
  AND d.source_document_id = {{ Number(appsmith.store.currentSalesOrderShippingId || SalesOrderTable.selectedRow.documentId || 0) }}
ORDER BY d.id DESC
LIMIT 1;
