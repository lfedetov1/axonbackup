SELECT
    di.id AS documentItemId,
    di.line_no AS lineNo,

    p.id AS productId,
    p.code AS productCode,
    p.name AS productName,
    p.barcode,
    p.sku,
    p.track_stock AS trackStock,

    di.description,
    di.quantity,
    COALESCE(di.reserved_quantity, 0) AS reservedQuantity,

    u.id AS unitId,
    u.code AS unitCode,

    di.unit_price AS unitPrice,
    di.discount_amount AS discountAmount,
    di.line_subtotal AS lineSubtotal,

    tr.id AS taxRateId,
    tr.rate AS taxRate,

    di.tax_amount AS taxAmount,
    di.line_total AS lineTotal,
    di.note,

    0 AS availableStock
FROM document_items di
LEFT JOIN products p ON p.id = di.product_id
LEFT JOIN units_of_measure u ON u.id = di.unit_id
LEFT JOIN tax_rates tr ON tr.id = di.tax_rate_id
WHERE di.document_id = {{this.params.documentId}}
ORDER BY di.line_no;
