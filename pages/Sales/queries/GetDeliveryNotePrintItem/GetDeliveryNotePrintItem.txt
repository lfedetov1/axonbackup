SELECT
    di.id AS documentItemId,
    di.line_no AS lineNo,
    p.id AS productId,
    p.code AS productCode,
    p.name AS productName,
    p.barcode AS barcode,
    di.description AS description,
    di.quantity AS quantity,
    u.code AS unitCode,
    di.note AS note

FROM document_items di
LEFT JOIN products p ON p.id = di.product_id
LEFT JOIN units_of_measure u ON u.id = di.unit_id

WHERE di.document_id = {{ Number(appsmith.store.selectedDeliveryNotePrintId || 0) }}

ORDER BY di.line_no;
