SELECT CONCAT(
    'DN-',
    DATE_FORMAT(CURDATE(), '%Y%m%d'),
    '-',
    LPAD(
      COALESCE((
        SELECT COUNT(*) + 1
        FROM documents d
        WHERE d.company_id = {{ Number(appsmith.store.companyId || 0) }}
          AND d.document_type = 'DELIVERY_NOTE'
          AND DATE(d.created_at) = CURDATE()
      ), 1),
      4,
      '0'
    )
) AS nextDeliveryNoteNumber;
