INSERT INTO documents (
    company_id,
    document_type,
    document_number,
    document_date,
    due_date,
    partner_id,
    warehouse_id,
    payment_method,
    currency_code,
    exchange_rate,
    subtotal_amount,
    discount_amount,
    tax_amount,
    total_amount,
    paid_amount,
    status,
    posting_status,
    sales_channel,
    source_document_id,
    note,
    created_by_user_id
)
SELECT
    q.company_id,
    'SALES_INVOICE',
    {{ this.params.invoiceNumber }},
    CURDATE(),
    q.due_date,
    q.partner_id,
    q.warehouse_id,
    q.payment_method,
    q.currency_code,
    q.exchange_rate,
    q.subtotal_amount,
    q.discount_amount,
    q.tax_amount,
    q.total_amount,
    0,
    'DRAFT',
    'NOT_POSTED',
    q.sales_channel,
    q.id,
    CONCAT('Created from quote ', q.document_number),
    {{ Number(appsmith.store.userId || q.created_by_user_id || 0) || null }}
FROM documents q
WHERE q.id = {{ Number(this.params.quoteId || this.params.documentId || 0) }}
  AND q.company_id = {{ Number(appsmith.store.companyId || 0) }}
  AND q.document_type = 'QUOTE'
  AND q.status NOT IN ('CANCELLED', 'CONVERTED')
  AND NOT EXISTS (
      SELECT 1
      FROM documents inv
      WHERE inv.company_id = q.company_id
        AND inv.document_type = 'SALES_INVOICE'
        AND inv.source_document_id = q.id
        AND inv.status <> 'CANCELLED'
  );
