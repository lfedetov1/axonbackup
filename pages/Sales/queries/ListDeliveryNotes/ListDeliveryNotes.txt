SELECT
    d.id AS documentId,
    d.document_number AS `Delivery Number`,
    d.document_date AS `Delivery Date`,
    d.status AS `Status`,
    d.posting_status AS `Posting`,

    so.document_number AS `Sales Order`,

    COALESCE(bp.legal_name, bp.name) AS `Customer`,
    bp.code AS `Customer Code`,

    w.code AS `Warehouse Code`,
    w.name AS `Warehouse`,

    d.sales_channel AS `Channel`,
    d.fulfillment_type AS `Fulfillment`,

    COUNT(di.id) AS `Lines`,
    COALESCE(SUM(di.quantity), 0) AS `Quantity`,

    u.username AS `Created By`,
    d.created_at AS `Created At`

FROM documents d
LEFT JOIN documents so ON so.id = d.source_document_id
LEFT JOIN business_partners bp ON bp.id = d.partner_id
LEFT JOIN warehouses w ON w.id = d.warehouse_id
LEFT JOIN users u ON u.id = d.created_by_user_id
LEFT JOIN document_items di ON di.document_id = d.id

WHERE d.company_id = {{ Number(appsmith.store.companyId || 0) }}
  AND d.document_type = 'DELIVERY_NOTE'
  AND (
      {{ DeliveryNoteSearchInput.text || "" }} = ""
      OR d.document_number LIKE CONCAT('%', {{ DeliveryNoteSearchInput.text || "" }}, '%')
      OR so.document_number LIKE CONCAT('%', {{ DeliveryNoteSearchInput.text || "" }}, '%')
      OR bp.name LIKE CONCAT('%', {{ DeliveryNoteSearchInput.text || "" }}, '%')
      OR bp.legal_name LIKE CONCAT('%', {{ DeliveryNoteSearchInput.text || "" }}, '%')
      OR bp.code LIKE CONCAT('%', {{ DeliveryNoteSearchInput.text || "" }}, '%')
  )
  AND d.document_date >= {{ DeliveryNoteDateFrom.selectedDate ? moment(DeliveryNoteDateFrom.selectedDate).format("YYYY-MM-DD") : moment().startOf("month").format("YYYY-MM-DD") }}
  AND d.document_date <= {{ DeliveryNoteDateTo.selectedDate ? moment(DeliveryNoteDateTo.selectedDate).format("YYYY-MM-DD") : moment().format("YYYY-MM-DD") }}

GROUP BY
    d.id,
    d.document_number,
    d.document_date,
    d.status,
    d.posting_status,
    so.document_number,
    bp.legal_name,
    bp.name,
    bp.code,
    w.code,
    w.name,
    d.sales_channel,
    d.fulfillment_type,
    u.username,
    d.created_at

ORDER BY d.document_date DESC, d.id DESC;
