SELECT
  d.id AS documentId,
  d.document_number AS documentNumber,
  d.document_date AS documentDate,
  d.status AS status,
  d.source_warehouse_id AS sourceWarehouseId,
  d.destination_warehouse_id AS destinationWarehouseId,
  d.note AS note
FROM documents d
WHERE d.id = {{this.params.documentId}}
  AND d.company_id = {{appsmith.store.companyId}}
  AND d.document_type = 'TRANSFER_REQUEST'
LIMIT 1;
