SELECT
    d.id AS documentId,
    d.document_number AS documentNumber,
    d.document_date AS documentDate,
    d.status AS status,
    d.note AS note,
    d.created_at AS createdAt,
    d.updated_at AS updatedAt,

    w.id AS warehouseId,
    w.code AS warehouseCode,
    w.name AS warehouseName,

    c.name AS companyName,
    c.tax_number AS companyTaxNumber,
    c.registration_number AS companyRegistrationNumber,
    c.email AS companyEmail,
    c.phone AS companyPhone,
    c.address_line1 AS companyAddress,
    c.city AS companyCity,
    c.postal_code AS companyPostalCode,
    c.logo_path AS company_logo_path,

    u.username AS createdBy,
    au.username AS approvedBy,

    COUNT(snapshot.id) AS lineCount,
    COALESCE(SUM(snapshot.system_quantity), 0) AS totalSystemQuantity,
    COALESCE(SUM(COALESCE(counted.counted_quantity, 0)), 0) AS totalCountedQuantity,
    COALESCE(SUM(COALESCE(counted.counted_quantity, 0) - COALESCE(snapshot.system_quantity, 0)), 0) AS totalVarianceQuantity,
    COALESCE(SUM((COALESCE(counted.counted_quantity, 0) - COALESCE(snapshot.system_quantity, 0)) * COALESCE(snapshot.unit_price, 0)), 0) AS totalVarianceValue

FROM documents d
JOIN companies c ON c.id = d.company_id
LEFT JOIN warehouses w ON w.id = d.warehouse_id
LEFT JOIN users u ON u.id = d.created_by_user_id
LEFT JOIN users au ON au.id = d.approved_by_user_id
LEFT JOIN document_items snapshot ON snapshot.document_id = d.id

LEFT JOIN (
    SELECT
        d2.source_document_id AS inventoryCountId,
        di2.product_id,
        SUM(di2.counted_quantity) AS counted_quantity
    FROM documents d2
    JOIN document_items di2 ON di2.document_id = d2.id
    WHERE d2.company_id = {{appsmith.store.companyId}}
      AND d2.document_type = 'INVENTORY_COUNT_DOCUMENT'
      AND d2.source_document_id = {{this.params.inventoryCountId || appsmith.store.currentInventoryCountId || 0}}
    GROUP BY
        d2.source_document_id,
        di2.product_id
) counted
  ON counted.inventoryCountId = d.id
 AND counted.product_id = snapshot.product_id

WHERE d.id = {{this.params.inventoryCountId || appsmith.store.currentInventoryCountId || InventoryCountTable.selectedRow.documentId || 0}}
  AND d.company_id = {{appsmith.store.companyId}}
  AND d.document_type = 'INVENTORY_COUNT'

GROUP BY
    d.id,
    d.document_number,
    d.document_date,
    d.status,
    d.note,
    d.created_at,
    d.updated_at,
    w.id,
    w.code,
    w.name,
    c.name,
    c.tax_number,
    c.registration_number,
    c.email,
    c.phone,
    c.address_line1,
    c.city,
    c.postal_code,
    c.logo_path,
    u.username,
    au.username

LIMIT 1;
