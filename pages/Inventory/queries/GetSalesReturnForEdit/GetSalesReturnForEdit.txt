SELECT
    d.id AS documentId,
    d.document_number AS documentNumber,
    d.document_date AS documentDate,
    d.status AS status,
    d.source_document_id AS sourceDocumentId,
    src.document_number AS sourceDocumentNumber,
    src.document_type AS sourceDocumentType,
    d.partner_id AS customerId,
    d.warehouse_id AS warehouseId,
    d.note AS note
FROM documents d
LEFT JOIN documents src ON src.id = d.source_document_id
WHERE d.id = {{this.params.documentId}}
  AND d.company_id = {{appsmith.store.companyId}}
  AND d.document_type = 'SALES_RETURN'
LIMIT 1;
