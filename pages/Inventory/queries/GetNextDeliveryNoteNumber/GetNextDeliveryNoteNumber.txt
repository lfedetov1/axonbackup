SELECT
    CONCAT(
        'DN-',
        DATE_FORMAT(CURDATE(), '%Y'),
        '-',
        LPAD(
            COALESCE(
                MAX(
                    CAST(
                        SUBSTRING_INDEX(document_number, '-', -1) AS UNSIGNED
                    )
                ),
                0
            ) + 1,
            5,
            '0'
        )
    ) AS nextNumber
FROM documents
WHERE company_id = {{appsmith.store.companyId}}
  AND document_type = 'DELIVERY_NOTE'
  AND document_number LIKE CONCAT('DN-', DATE_FORMAT(CURDATE(), '%Y'), '-%');
