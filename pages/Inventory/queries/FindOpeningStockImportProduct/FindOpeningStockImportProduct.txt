SELECT
    p.id AS productId,
    p.code AS productCode,
    p.name AS productName,
    p.barcode AS barcode,
    p.sku AS sku,
    p.unit_id AS unitId,
    u.code AS unitCode,
    COALESCE(pp.purchase_price, 0) AS purchasePrice
FROM products p
LEFT JOIN units_of_measure u
  ON u.id = p.unit_id
LEFT JOIN product_prices pp
  ON pp.product_id = p.id
 AND pp.price_type = 'PURCHASE'
 AND pp.valid_to IS NULL
WHERE p.company_id = {{appsmith.store.companyId}}
  AND p.is_active = 1
  AND (
      p.barcode = {{this.params.lookup}}
      OR p.code = {{this.params.lookup}}
      OR p.sku = {{this.params.lookup}}
  )
LIMIT 1;
