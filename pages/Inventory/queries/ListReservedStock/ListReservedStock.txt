SELECT
    d.id AS documentId,
    d.document_number AS `Document Number`,
    d.document_type AS `Document Type`,
    d.document_date AS `Document Date`,
    d.status AS `Status`,

    w.code AS `Warehouse Code`,
    w.name AS `Warehouse`,

    p.id AS productId,
    p.code AS `Product Code`,
    p.name AS `Product Name`,
    p.barcode AS `Barcode`,
    p.sku AS `SKU`,

    u.code AS `Unit`,

    SUM(COALESCE(di.quantity, 0)) AS `Reserved Quantity`,

    COALESCE(di.unit_price, 0) AS `Unit Price`,

    SUM(COALESCE(di.quantity, 0) * COALESCE(di.unit_price, 0)) AS `Reserved Value`,

    COALESCE(ib.quantity_balance, 0) AS `Available Stock`,

    CASE
      WHEN COALESCE(ib.quantity_balance, 0) >= SUM(COALESCE(di.quantity, 0)) THEN 'OK'
      ELSE 'SHORT'
    END AS `Stock Status`

FROM documents d

JOIN document_items di
  ON di.document_id = d.id

JOIN products p
  ON p.id = di.product_id

LEFT JOIN units_of_measure u
  ON u.id = di.unit_id

LEFT JOIN warehouses w
  ON w.id = d.warehouse_id

LEFT JOIN v_inventory_balance ib
  ON ib.company_id = d.company_id
 AND ib.warehouse_id = d.warehouse_id
 AND ib.product_id = di.product_id

WHERE d.company_id = {{appsmith.store.companyId}}
  AND d.status IN ('DRAFT', 'CONFIRMED', 'APPROVED', 'PICKED')
  AND d.document_type IN (
      'SALES_ORDER',
      'DELIVERY_NOTE',
      'TRANSFER_REQUEST'
  )
  AND COALESCE(di.quantity, 0) > 0
  AND (
      {{InventorySearchInput.text || ''}} = ''
      OR d.document_number LIKE CONCAT('%', {{InventorySearchInput.text || ''}}, '%')
      OR p.code LIKE CONCAT('%', {{InventorySearchInput.text || ''}}, '%')
      OR p.name LIKE CONCAT('%', {{InventorySearchInput.text || ''}}, '%')
      OR p.barcode LIKE CONCAT('%', {{InventorySearchInput.text || ''}}, '%')
      OR p.sku LIKE CONCAT('%', {{InventorySearchInput.text || ''}}, '%')
  )
  AND (
      {{Number(InventoryWarehouseSelect.selectedOptionValue || appsmith.store.warehouseId || 0)}} = 0
      OR d.warehouse_id = {{Number(InventoryWarehouseSelect.selectedOptionValue || appsmith.store.warehouseId || 0)}}
  )
  AND d.document_date >= {{moment(InventoryDateFrom.selectedDate || moment().startOf("month")).format("YYYY-MM-DD")}}
  AND d.document_date <= {{moment(InventoryDateTo.selectedDate || moment()).format("YYYY-MM-DD")}}

GROUP BY
    d.id,
    d.document_number,
    d.document_type,
    d.document_date,
    d.status,
    w.code,
    w.name,
    p.id,
    p.code,
    p.name,
    p.barcode,
    p.sku,
    u.code,
    di.unit_price,
    ib.quantity_balance

ORDER BY
    d.document_date DESC,
    d.id DESC,
    p.name ASC;
