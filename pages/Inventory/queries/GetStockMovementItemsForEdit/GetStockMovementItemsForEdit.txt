SELECT
    di.line_no AS lineNo,
    p.id AS productId,
    p.code AS productCode,
    p.name AS productName,
    p.barcode AS barcode,
    p.sku AS sku,
    di.unit_id AS unitId,
    u.code AS unitCode,
    di.quantity AS quantity,
    di.unit_price AS unitCost,
    di.line_total AS lineTotal,
    di.batch_number AS batchNumber,
    di.serial_number AS serialNumber,
    di.note AS note,
    COALESCE(ib.quantity_balance, 0) AS currentStock
FROM document_items di
JOIN documents d
  ON d.id = di.document_id
JOIN products p
  ON p.id = di.product_id
LEFT JOIN units_of_measure u
  ON u.id = di.unit_id
LEFT JOIN v_inventory_balance ib
  ON ib.company_id = d.company_id
 AND ib.product_id = p.id
 AND ib.warehouse_id = COALESCE(d.source_warehouse_id, d.warehouse_id, d.destination_warehouse_id)
WHERE d.company_id = {{appsmith.store.companyId}}
  AND d.document_number = {{MovementNumberInput.text.trim()}}
ORDER BY di.line_no ASC;
