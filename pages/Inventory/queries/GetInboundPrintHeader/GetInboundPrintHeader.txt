SELECT
    d.id AS documentId,
    d.document_number AS inboundNumber,
    d.document_date AS documentDate,
    d.expected_date AS expectedDate,
    d.status AS status,
    d.reference_number AS referenceNumber,
    d.note AS note,

    bp.code AS supplierCode,
    bp.name AS supplierName,
    bp.tax_number AS supplierTaxNumber,
    bp.email AS supplierEmail,
    bp.phone AS supplierPhone,

    w.code AS warehouseCode,
    w.name AS warehouseName,

    c.name AS companyName,
    c.legal_name AS companyLegalName,
    c.address_line1 AS companyAddressLine1,
    c.postal_code AS companyPostalCode,
    c.city AS companyCity,
    c.country_code AS companyCountryCode,
    c.tax_number AS companyTaxNumber,
    c.email AS companyEmail,
    c.phone AS companyPhone,
    c.logo_path AS companyLogoPath,

    {{appsmith.store.username || ''}} AS printedBy

FROM documents d

LEFT JOIN business_partners bp
  ON bp.id = d.partner_id

LEFT JOIN warehouses w
  ON w.id = d.warehouse_id

LEFT JOIN companies c
  ON c.id = d.company_id

WHERE d.id = {{this.params.documentId}}
  AND d.document_type = 'INBOUND_DELIVERY'
LIMIT 1;
