SELECT
    COALESCE(SUM(COALESCE(stock.current_stock, 0)), 0) AS currentStockQuantity,
    COALESCE(SUM(COALESCE(stock.current_stock_value, 0)), 0) AS currentStockValue,

    COALESCE(sold.sold_quantity, 0) AS soldQuantity,
    COALESCE(sold.sold_cost_value, 0) AS soldCostValue,

    COALESCE(SUM(
        CASE
            WHEN p.track_stock = 1
             AND p.is_active = 1
             AND p.minimum_stock > 0
             AND COALESCE(stock.current_stock, 0) < p.minimum_stock
            THEN 1
            ELSE 0
        END
    ), 0) AS lowStockItemCount

FROM products p

JOIN warehouses w
  ON w.company_id = p.company_id
 AND w.is_active = 1

LEFT JOIN (
    SELECT
        ib.company_id,
        ib.warehouse_id,
        ib.product_id,
        ib.quantity_balance AS current_stock,
        ib.cost_balance AS current_stock_value
    FROM v_inventory_balance ib
) stock
  ON stock.company_id = p.company_id
 AND stock.warehouse_id = w.id
 AND stock.product_id = p.id

LEFT JOIN (
    SELECT
        sm.company_id,
        SUM(sm.quantity) AS sold_quantity,
        SUM(sm.total_cost) AS sold_cost_value
    FROM stock_movements sm
    LEFT JOIN documents d ON d.id = sm.document_id
    WHERE sm.company_id = {{appsmith.store.companyId}}
      AND sm.movement_type = 'OUT'
      AND (
          {{appsmith.store.warehouseId || 0}} = 0
          OR sm.warehouse_id = {{appsmith.store.warehouseId || 0}}
      )
      AND (
          d.document_type IN ('SALES_INVOICE', 'DELIVERY_NOTE')
          OR COALESCE(d.document_type, '') = ''
      )
      AND sm.movement_date >= {{InventoryDateFrom.selectedDate || moment().startOf("month").format("YYYY-MM-DD")}}
      AND sm.movement_date < DATE_ADD({{InventoryDateTo.selectedDate || moment().format("YYYY-MM-DD")}}, INTERVAL 1 DAY)
    GROUP BY sm.company_id
) sold
  ON sold.company_id = p.company_id

WHERE p.company_id = {{appsmith.store.companyId}}
  AND p.track_stock = 1
  AND p.is_active = 1
  AND (
      {{appsmith.store.warehouseId || 0}} = 0
      OR w.id = {{appsmith.store.warehouseId || 0}}
  );
