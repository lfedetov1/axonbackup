SELECT
    p.id AS productId,
    p.code AS productCode,
    p.name AS productName,

    u.code AS unitCode,
    u.name AS unitName,

    SUM(sm.quantity) AS soldQuantity,
    SUM(sm.total_cost) AS soldCostValue,

    COUNT(sm.id) AS movementCount,

    MAX(sm.movement_date) AS lastSoldDate

FROM stock_movements sm

JOIN products p
  ON p.id = sm.product_id

LEFT JOIN units_of_measure u
  ON u.id = p.unit_id

LEFT JOIN documents d
  ON d.id = sm.document_id

WHERE sm.company_id = {{appsmith.store.companyId}}
  AND sm.movement_type = 'OUT'
  AND (
      {{InventoryWarehouseSelect.selectedOptionValue || appsmith.store.warehouseId || 0}} = 0
      OR sm.warehouse_id = {{InventoryWarehouseSelect.selectedOptionValue || appsmith.store.warehouseId || 0}}
  )
  AND (
      d.document_type IN ('SALES_INVOICE', 'DELIVERY_NOTE')
      OR COALESCE(d.document_type, '') = ''
  )
  AND sm.movement_date >= {{InventoryDateFrom.selectedDate || moment().startOf("month").format("YYYY-MM-DD")}}
  AND sm.movement_date < DATE_ADD({{InventoryDateTo.selectedDate || moment().format("YYYY-MM-DD")}}, INTERVAL 1 DAY)
  AND (
      {{!InventorySearchInput.text}}
      OR p.code LIKE CONCAT('%', {{InventorySearchInput.text}}, '%')
      OR p.name LIKE CONCAT('%', {{InventorySearchInput.text}}, '%')
  )

GROUP BY
    p.id,
    p.code,
    p.name,
    u.code,
    u.name

ORDER BY
    soldQuantity DESC,
    soldCostValue DESC,
    p.name ASC

LIMIT 20;
