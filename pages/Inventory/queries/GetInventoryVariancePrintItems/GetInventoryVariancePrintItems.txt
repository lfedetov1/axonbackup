SELECT
    di.id AS documentItemId,
    di.line_no AS lineNo,

    p.id AS productId,
    p.code AS productCode,
    p.name AS productName,
    p.barcode AS barcode,
    p.sku AS sku,

    u.code AS unitCode,

    COALESCE(di.system_quantity, 0) AS systemQuantity,
    COALESCE(di.counted_quantity, 0) AS countedQuantity,
    COALESCE(di.variance_quantity, 0) AS varianceQuantity,
    COALESCE(di.unit_price, 0) AS unitCost,
    COALESCE(di.variance_value, 0) AS varianceValue,

    wl.code AS locationCode,

    di.batch_number AS batchNumber,
    di.serial_number AS serialNumber,
    di.note AS note,

    cb.username AS countedBy,
    di.counted_at AS countedAt

FROM document_items di
JOIN products p ON p.id = di.product_id
LEFT JOIN units_of_measure u ON u.id = di.unit_id
LEFT JOIN warehouse_locations wl ON wl.id = di.warehouse_location_id
LEFT JOIN users cb ON cb.id = di.counted_by_user_id

WHERE di.document_id = {{this.params.documentId || appsmith.store.currentInventoryCountId || InventoryCountTable.selectedRow.documentId || 0}}
  AND COALESCE(di.variance_quantity, 0) <> 0

ORDER BY
    ABS(COALESCE(di.variance_value, 0)) DESC,
    di.line_no;
