SELECT
    di.line_no AS lineNo,

    p.code AS productCode,
    p.name AS productName,
    p.barcode AS barcode,
    p.sku AS sku,

    di.description AS description,

    COALESCE(di.expected_quantity, di.quantity, 0) AS expectedQty,
    COALESCE(di.received_quantity, 0) AS receivedQty,
    COALESCE(di.variance_quantity, 0) AS varianceQty,

    u.code AS unitCode,

    di.unit_price AS unitCost,
    di.line_total AS lineTotal,
    di.note AS note

FROM document_items di

JOIN products p
  ON p.id = di.product_id

LEFT JOIN units_of_measure u
  ON u.id = di.unit_id

WHERE di.document_id = {{this.params.documentId}}

ORDER BY
    di.line_no;
