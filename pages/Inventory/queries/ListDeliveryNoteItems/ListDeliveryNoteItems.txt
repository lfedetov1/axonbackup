SELECT
    di.id AS documentItemId,
    di.document_id AS documentId,
    di.line_no AS `Line No`,

    p.id AS productId,
    p.code AS `Product Code`,
    p.name AS `Product Name`,
    p.barcode AS `Barcode`,
    p.sku AS `SKU`,

    di.description AS `Description`,
    di.quantity AS `Quantity`,

    u.id AS unitId,
    u.code AS `Unit`,

    di.unit_price AS `Unit Price`,
    di.line_total AS `Line Total`,
    di.note AS `Note`

FROM document_items di

JOIN products p
  ON p.id = di.product_id

LEFT JOIN units_of_measure u
  ON u.id = di.unit_id

WHERE di.document_id = {{this.params.documentId || DeliveryNotesTable.selectedRow.documentId || 0}}

ORDER BY
    di.line_no;
