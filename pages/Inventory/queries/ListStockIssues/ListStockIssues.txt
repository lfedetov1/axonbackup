SELECT
    d.id AS `Issue ID`,
    d.document_number AS `Issue Number`,
    d.document_date AS `Issue Date`,
    d.status AS `Status`,

    w.code AS `Warehouse Code`,
    w.name AS `Warehouse`,

    COUNT(di.id) AS `Lines`,
    COALESCE(SUM(di.quantity), 0) AS `Total Quantity`,
    d.total_amount AS `Total Value`,

    d.note AS `Reason / Note`,
    d.created_at AS `Created At`,
    d.updated_at AS `Updated At`

FROM documents d

LEFT JOIN warehouses w
  ON w.id = d.warehouse_id

LEFT JOIN document_items di
  ON di.document_id = d.id

WHERE d.document_type = 'STOCK_OUT'

AND (
    {{ StockIssueSearchInput.text || "" }} = ""
    OR d.document_number LIKE CONCAT('%', {{ StockIssueSearchInput.text || "" }}, '%')
    OR d.note LIKE CONCAT('%', {{ StockIssueSearchInput.text || "" }}, '%')
)

AND d.document_date >= {{ moment(StockIssueDateFrom.selectedDate || moment().startOf("month")).format("YYYY-MM-DD") }}
AND d.document_date <= {{ moment(StockIssueDateTo.selectedDate || moment()).format("YYYY-MM-DD") }}

AND (
    {{ AppAccess.canViewAllWarehouses() ? 1 : 0 }} = 1
    OR d.warehouse_id IN (
        {{ (appsmith.store.warehouseAccessIds || [0]).join(",") }}
    )
)

GROUP BY
    d.id,
    d.document_number,
    d.document_date,
    d.status,
    w.code,
    w.name,
    d.total_amount,
    d.note,
    d.created_at,
    d.updated_at

ORDER BY
    d.id DESC;
