SELECT
    di.id AS documentItemId,
    d.id AS documentId,
    d.document_number AS `Receipt Number`,
    d.warehouse_id AS warehouseId,

    p.id AS productId,
    p.code AS `Product Code`,
    p.name AS `Product Name`,
    p.barcode AS `Barcode`,
    p.sku AS `SKU`,

    u.code AS `Unit`,

    di.quantity AS `Received Quantity`,

    COALESCE((
        SELECT COUNT(*)
        FROM warehouse_location_products wlp
        WHERE wlp.company_id = d.company_id
          AND wlp.warehouse_id = d.warehouse_id
          AND wlp.product_id = p.id
          AND wlp.is_active = 1
    ), 0) AS `Mapped Locations`,

    di.note AS `Note`

FROM document_items di

JOIN documents d
  ON d.id = di.document_id

JOIN products p
  ON p.id = di.product_id

LEFT JOIN units_of_measure u
  ON u.id = di.unit_id

WHERE d.company_id = {{appsmith.store.companyId}}
  AND di.document_id = {{PutawayPendingReceiptsTable.selectedRow.documentId || 0}}

ORDER BY
    di.line_no,
    p.name;
