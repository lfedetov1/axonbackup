SELECT
    d.id AS documentId,
    d.document_number AS `Inbound Number`,
    d.document_date AS `Document Date`,
    d.expected_date AS `Expected Date`,
    d.status AS `Status`,

    bp.code AS `Supplier Code`,
    bp.name AS `Supplier`,

    w.code AS `Warehouse Code`,
    w.name AS `Warehouse`,

    COUNT(di.id) AS `Line Count`,
    COALESCE(SUM(COALESCE(di.expected_quantity, di.quantity)), 0) AS `Expected Quantity`,
    COALESCE(SUM(COALESCE(di.received_quantity, 0)), 0) AS `Received Quantity`,
    COALESCE(SUM(COALESCE(di.variance_quantity, 0)), 0) AS `Variance Quantity`,

    d.reference_number AS `Reference Number`,
    d.note AS `Note`,
    d.created_at AS `Created At`,
    d.updated_at AS `Updated At`

FROM documents d

LEFT JOIN business_partners bp
  ON bp.id = d.partner_id

LEFT JOIN warehouses w
  ON w.id = d.warehouse_id

LEFT JOIN document_items di
  ON di.document_id = d.id

WHERE d.company_id = {{appsmith.store.companyId}}
  AND d.document_type = 'INBOUND_DELIVERY'
  AND (
      {{InventorySearchInput.text || ''}} = ''
      OR d.document_number LIKE CONCAT('%', {{InventorySearchInput.text || ''}}, '%')
      OR bp.name LIKE CONCAT('%', {{InventorySearchInput.text || ''}}, '%')
      OR bp.code LIKE CONCAT('%', {{InventorySearchInput.text || ''}}, '%')
      OR d.reference_number LIKE CONCAT('%', {{InventorySearchInput.text || ''}}, '%')
  )
  AND (
      {{Number(InventoryWarehouseSelect.selectedOptionValue || 0)}} = 0
      OR d.warehouse_id = {{Number(InventoryWarehouseSelect.selectedOptionValue || 0)}}
  )
  AND d.document_date >= {{moment(InventoryDateFrom.selectedDate || '2000-01-01').format("YYYY-MM-DD")}}
  AND d.document_date <= {{moment(InventoryDateTo.selectedDate || '2099-12-31').format("YYYY-MM-DD")}}

GROUP BY
    d.id,
    d.document_number,
    d.document_date,
    d.expected_date,
    d.status,
    bp.code,
    bp.name,
    w.code,
    w.name,
    d.reference_number,
    d.note,
    d.created_at,
    d.updated_at

ORDER BY
    d.document_date DESC,
    d.id DESC;
