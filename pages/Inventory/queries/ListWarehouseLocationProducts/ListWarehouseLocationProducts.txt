SELECT
    wlp.id AS mappingId,
    wlp.location_id AS locationId,
    wl.code AS locationCode,
    wl.name AS locationName,

    p.id AS productId,
    p.code AS productCode,
    p.name AS productName,
    p.barcode AS barcode,
    p.sku AS sku,

    u.code AS unitCode,

    wlp.is_primary_location AS isPrimaryLocation,
    wlp.minimum_quantity AS minimumQuantity,
    wlp.maximum_quantity AS maximumQuantity,
    wlp.reorder_quantity AS reorderQuantity,
    wlp.note AS note,
    wlp.is_active AS isActive,

    COALESCE(ib.quantity_balance, 0) AS warehouseStock

FROM warehouse_location_products wlp

JOIN warehouse_locations wl
  ON wl.id = wlp.location_id

JOIN products p
  ON p.id = wlp.product_id

LEFT JOIN units_of_measure u
  ON u.id = p.unit_id

LEFT JOIN v_inventory_balance ib
  ON ib.company_id = wlp.company_id
 AND ib.warehouse_id = wlp.warehouse_id
 AND ib.product_id = wlp.product_id

WHERE wlp.company_id = {{appsmith.store.companyId}}
  AND wlp.location_id = {{this.params.locationId || appsmith.store.selectedWarehouseLocationId || 0}}
  AND wlp.is_active = 1

ORDER BY
    wlp.is_primary_location DESC,
    p.name ASC;
