SELECT
    d.id AS documentId,
    d.document_number AS documentNumber,
    d.document_type AS documentType,
    d.status AS status,
    COUNT(di.id) AS lineCount,
    SUM(CASE WHEN COALESCE(di.picked_quantity, 0) >= di.quantity THEN 1 ELSE 0 END) AS pickedLineCount,
    COALESCE(SUM(di.quantity), 0) AS requiredQuantity,
    COALESCE(SUM(di.picked_quantity), 0) AS pickedQuantity
FROM documents d
LEFT JOIN document_items di ON di.document_id = d.id
WHERE d.id = {{this.params.documentId}}
  AND d.company_id = {{appsmith.store.companyId}}
GROUP BY
    d.id,
    d.document_number,
    d.document_type,
    d.status;
