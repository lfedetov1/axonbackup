SELECT
    r.code AS roleCode,
    r.name AS roleName,
    p.code AS permissionCode
FROM company_users cu
JOIN roles r
  ON r.id = cu.role_id
JOIN role_permissions rp
  ON rp.role_id = r.id
JOIN permissions p
  ON p.id = rp.permission_id
JOIN users u
  ON u.id = cu.user_id
WHERE cu.company_id = {{appsmith.store.companyId}}
  AND cu.user_id = {{appsmith.store.userId}}
  AND u.is_active = 1
ORDER BY
    p.code;
