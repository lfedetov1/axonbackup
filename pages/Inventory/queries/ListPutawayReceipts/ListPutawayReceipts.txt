SELECT
    d.id AS documentId,
    d.document_number AS `Receipt Number`,
    d.document_date AS `Receipt Date`,
    bp.name AS `Supplier`,
    w.code AS `Warehouse Code`,
    w.name AS `Warehouse`,
    COUNT(di.id) AS `Line Count`,
    COALESCE(SUM(di.quantity), 0) AS `Total Quantity`,
    d.status AS `Status`,
    d.note AS `Note`
FROM documents d
LEFT JOIN business_partners bp
  ON bp.id = d.partner_id
LEFT JOIN warehouses w
  ON w.id = d.warehouse_id
LEFT JOIN document_items di
  ON di.document_id = d.id
WHERE d.company_id = {{appsmith.store.companyId}}
  AND d.document_type = 'PURCHASE_RECEIPT'
  AND d.status IN ('POSTED', 'APPROVED')
  AND (
    {{ PutawaySearchInput.text || "" }} = ""
    OR wt.task_number LIKE CONCAT('%', {{ PutawaySearchInput.text || "" }}, '%')
    OR d.document_number LIKE CONCAT('%', {{ PutawaySearchInput.text || "" }}, '%')
)

AND wt.created_at >= {{ moment(PutawayDateFrom.selectedDate || moment().startOf("month")).format("YYYY-MM-DD") }}
AND wt.created_at < DATE_ADD({{ moment(PutawayDateTo.selectedDate || moment()).format("YYYY-MM-DD") }}, INTERVAL 1 DAY)

AND (
    {{ AppAccess.canViewAllWarehouses() ? 1 : 0 }} = 1
    OR wt.warehouse_id IN (
        {{ (appsmith.store.warehouseAccessIds || [0]).join(",") }}
    )
)
GROUP BY
    d.id,
    d.document_number,
    d.document_date,
    bp.name,
    w.code,
    w.name,
    d.status,
    d.note
ORDER BY
    d.document_date DESC,
    d.id DESC;
