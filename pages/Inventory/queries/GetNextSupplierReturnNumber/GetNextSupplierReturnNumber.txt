SELECT CONCAT('PR-', YEAR(CURDATE()), '-', LPAD(COALESCE(MAX(CAST(SUBSTRING_INDEX(document_number, '-', -1) AS UNSIGNED)), 0) + 1, 5, '0')) AS nextSupplierReturnNumber
FROM documents
WHERE company_id = {{appsmith.store.companyId}}
  AND document_type = 'PURCHASE_RETURN'
  AND document_number LIKE CONCAT('PR-', YEAR(CURDATE()), '-%');
