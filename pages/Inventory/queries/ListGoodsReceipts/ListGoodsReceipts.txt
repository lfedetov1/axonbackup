SELECT
    d.id AS `Receipt ID`,
    d.company_id AS `Company ID`,
    d.document_number AS `Receipt Number`,
    d.document_date AS `Receipt Date`,
    d.status AS `Status`,

    bp.code AS `Supplier Code`,
    bp.name AS `Supplier`,

    w.code AS `Warehouse Code`,
    w.name AS `Warehouse`,

    COUNT(di.id) AS `Lines`,
    COALESCE(SUM(di.quantity), 0) AS `Total Quantity`,
    d.total_amount AS `Total Value`,

    d.note AS `Note`,
    d.created_at AS `Created At`,
    d.updated_at AS `Updated At`

FROM documents d

LEFT JOIN business_partners bp
  ON bp.id = d.partner_id

LEFT JOIN warehouses w
  ON w.id = d.warehouse_id

LEFT JOIN document_items di
  ON di.document_id = d.id

WHERE d.document_type = 'PURCHASE_RECEIPT'

AND (
    {{ GoodsReceiptSearchInput.text || "" }} = ""
    OR d.document_number LIKE CONCAT('%', {{ GoodsReceiptSearchInput.text || "" }}, '%')
    OR bp.name LIKE CONCAT('%', {{ GoodsReceiptSearchInput.text || "" }}, '%')
    OR bp.code LIKE CONCAT('%', {{ GoodsReceiptSearchInput.text || "" }}, '%')
)

AND d.document_date >= {{ moment(GoodsReceiptDateFrom.selectedDate || moment().startOf("month")).format("YYYY-MM-DD") }}
AND d.document_date <= {{ moment(GoodsReceiptDateTo.selectedDate || moment()).format("YYYY-MM-DD") }}

AND (
    {{ AppAccess.canViewAllWarehouses() ? 1 : 0 }} = 1
    OR d.warehouse_id IN (
        {{ (appsmith.store.warehouseAccessIds || [0]).join(",") }}
    )
)

GROUP BY
    d.id,
    d.company_id,
    d.document_number,
    d.document_date,
    d.status,
    bp.code,
    bp.name,
    w.code,
    w.name,
    d.total_amount,
    d.note,
    d.created_at,
    d.updated_at

ORDER BY
    d.id DESC;
