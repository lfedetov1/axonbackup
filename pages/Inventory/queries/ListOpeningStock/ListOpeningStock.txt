SELECT
    d.id AS documentId,
    d.document_number AS `Opening Stock Number`,
    d.document_date AS `Date`,
    d.status AS `Status`,

    w.code AS `Warehouse Code`,
    w.name AS `Warehouse`,

    COUNT(di.id) AS `Lines`,
    COALESCE(SUM(di.quantity), 0) AS `Total Quantity`,
    COALESCE(SUM(di.line_total), 0) AS `Total Value`,

    u.username AS `Created By`,
    d.note AS `Note`,
    d.created_at AS `Created At`,
    d.updated_at AS `Updated At`

FROM documents d
LEFT JOIN warehouses w ON w.id = d.warehouse_id
LEFT JOIN users u ON u.id = d.created_by_user_id
LEFT JOIN document_items di ON di.document_id = d.id

WHERE d.company_id = {{appsmith.store.companyId}}
  AND d.document_type = 'OPENING_STOCK'
	
	 AND (
      {{Number(InventoryWarehouseSelect.selectedOptionValue || appsmith.store.warehouseId || 0)}} = 0
      OR d.warehouse_id = {{Number(InventoryWarehouseSelect.selectedOptionValue || appsmith.store.warehouseId || 0)}}
  )
	
  AND (
    {{ OpeningStockSearchInput.text || "" }} = ""
    OR d.document_number LIKE CONCAT('%', {{ OpeningStockSearchInput.text || "" }}, '%')
    OR d.note LIKE CONCAT('%', {{ OpeningStockSearchInput.text || "" }}, '%')
)

AND d.document_date >= {{ moment(OpeningStockDateFrom.selectedDate || moment().startOf("month")).format("YYYY-MM-DD") }}
AND d.document_date <= {{ moment(OpeningStockDateTo.selectedDate || moment()).format("YYYY-MM-DD") }}

AND (
    {{ AppAccess.canViewAllWarehouses() ? 1 : 0 }} = 1
    OR d.warehouse_id IN (
        {{ (appsmith.store.warehouseAccessIds || [0]).join(",") }}
    )
)

GROUP BY
    d.id,
    d.document_number,
    d.document_date,
    d.status,
    w.code,
    w.name,
    u.username,
    d.note,
    d.created_at,
    d.updated_at

ORDER BY d.document_date DESC, d.id DESC;
