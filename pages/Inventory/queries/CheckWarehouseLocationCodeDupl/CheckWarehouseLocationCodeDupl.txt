SELECT
    id AS locationId,
    code AS locationCode,
    name AS locationName
FROM warehouse_locations
WHERE company_id = {{appsmith.store.companyId}}
  AND warehouse_id = {{StockLocationWarehouseSelect.selectedOptionValue}}
  AND code = {{StockLocationCodeInput.text.trim()}}
  AND id <> {{this.params.locationId || 0}}
LIMIT 1;
