SELECT
    w.id AS warehouseId,
    w.code AS warehouseCode,
    w.name AS warehouseName,

    p.id AS productId,
    p.code AS productCode,
    p.name AS productName,
    p.barcode AS barcode,
    p.sku AS sku,

    u.code AS unitCode,
    u.name AS unitName,

    COALESCE(ib.quantity_balance, 0) AS currentStock,
    p.minimum_stock AS minimumStock,
    p.reorder_level AS reorderLevel,
    p.maximum_stock AS maximumStock,

    GREATEST(p.minimum_stock - COALESCE(ib.quantity_balance, 0), 0) AS missingQuantity,

    COALESCE(p.purchase_price, 0) AS purchasePrice,
    GREATEST(p.minimum_stock - COALESCE(ib.quantity_balance, 0), 0)
      * COALESCE(p.purchase_price, 0) AS estimatedRestockValue,

    CASE
        WHEN COALESCE(ib.quantity_balance, 0) <= 0 THEN 'OUT_OF_STOCK'
        ELSE 'LOW_STOCK'
    END AS stockStatus,

    COALESCE(last_move.lastMovementDate, '') AS lastMovementDate

FROM products p

JOIN warehouses w
  ON w.company_id = p.company_id
 AND w.is_active = 1

LEFT JOIN units_of_measure u
  ON u.id = p.unit_id

LEFT JOIN v_inventory_balance ib
  ON ib.company_id = p.company_id
 AND ib.warehouse_id = w.id
 AND ib.product_id = p.id

LEFT JOIN (
    SELECT
        sm.company_id,
        sm.warehouse_id,
        sm.product_id,
        MAX(sm.movement_date) AS lastMovementDate
    FROM stock_movements sm
    WHERE sm.company_id = {{appsmith.store.companyId}}
    GROUP BY
        sm.company_id,
        sm.warehouse_id,
        sm.product_id
) last_move
  ON last_move.company_id = p.company_id
 AND last_move.warehouse_id = w.id
 AND last_move.product_id = p.id

WHERE p.company_id = {{appsmith.store.companyId}}
  AND p.track_stock = 1
  AND p.is_active = 1
  AND p.minimum_stock > 0
  AND COALESCE(ib.quantity_balance, 0) < p.minimum_stock
  AND (
      {{InventoryWarehouseSelect.selectedOptionValue || appsmith.store.warehouseId || 0}} = 0
      OR w.id = {{InventoryWarehouseSelect.selectedOptionValue || appsmith.store.warehouseId || 0}}
  )
  AND (
      {{!InventorySearchInput.text}}
      OR p.code LIKE CONCAT('%', {{InventorySearchInput.text}}, '%')
      OR p.name LIKE CONCAT('%', {{InventorySearchInput.text}}, '%')
      OR p.barcode LIKE CONCAT('%', {{InventorySearchInput.text}}, '%')
      OR p.sku LIKE CONCAT('%', {{InventorySearchInput.text}}, '%')
  )

ORDER BY
    missingQuantity DESC,
    estimatedRestockValue DESC,
    p.name ASC,
    w.name ASC;
