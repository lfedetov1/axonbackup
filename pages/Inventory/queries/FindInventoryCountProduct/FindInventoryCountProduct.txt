SELECT
    p.id AS productId,
    p.code AS productCode,
    p.name AS productName,
    p.barcode AS barcode,
    p.sku AS sku,
    p.unit_id AS unitId,
    u.code AS unitCode,

    COALESCE(ib.quantity_balance, 0) AS systemQuantity,
    COALESCE(pp.price, 0) AS unitCost,

    wl.id AS warehouseLocationId,
    wl.code AS locationCode

FROM products p
LEFT JOIN units_of_measure u ON u.id = p.unit_id

LEFT JOIN v_inventory_balance ib
  ON ib.company_id = {{appsmith.store.companyId}}
 AND ib.warehouse_id = {{this.params.warehouseId || appsmith.store.currentInventoryCountWarehouseId || 0}}
 AND ib.product_id = p.id

LEFT JOIN product_prices pp
  ON pp.product_id = p.id
 AND pp.price_type = 'PURCHASE'

LEFT JOIN warehouse_location_products wlp
  ON wlp.company_id = {{appsmith.store.companyId}}
 AND wlp.warehouse_id = {{this.params.warehouseId || appsmith.store.currentInventoryCountWarehouseId || 0}}
 AND wlp.product_id = p.id
 AND wlp.is_active = 1
 AND wlp.is_primary_location = 1

LEFT JOIN warehouse_locations wl
  ON wl.id = wlp.location_id

WHERE p.is_active = 1
  AND (
      p.barcode = {{this.params.lookup || ""}}
      OR p.code = {{this.params.lookup || ""}}
      OR p.sku = {{this.params.lookup || ""}}
      OR p.name LIKE CONCAT('%', {{this.params.lookup || ""}}, '%')
  )

ORDER BY
    CASE
      WHEN p.barcode = {{this.params.lookup || ""}} THEN 1
      WHEN p.code = {{this.params.lookup || ""}} THEN 2
      WHEN p.sku = {{this.params.lookup || ""}} THEN 3
      ELSE 4
    END,
    p.name

LIMIT 1;
