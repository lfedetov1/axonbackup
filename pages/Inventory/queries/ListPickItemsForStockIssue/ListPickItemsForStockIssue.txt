SELECT
    di.id AS documentItemId,
    d.id AS documentId,
    d.document_number AS `Issue Number`,
    d.warehouse_id AS warehouseId,

    p.id AS productId,
    p.code AS `Product Code`,
    p.name AS `Product Name`,
    p.barcode AS `Barcode`,
    p.sku AS `SKU`,

    u.code AS `Unit`,

    di.quantity AS `Required Quantity`,

    COALESCE(SUM(wpi.picked_quantity), 0) AS `Picked Quantity`,

    di.quantity - COALESCE(SUM(wpi.picked_quantity), 0) AS `Remaining Quantity`,

    COALESCE(ib.quantity_balance, 0) AS `Available Stock`,

    loc.id AS suggestedLocationId,
    loc.code AS `Suggested Location`,
    loc.name AS `Location Name`,

    CASE
        WHEN COALESCE(SUM(wpi.picked_quantity), 0) = 0 THEN 'READY'
        WHEN COALESCE(SUM(wpi.picked_quantity), 0) < di.quantity THEN 'PARTIAL'
        ELSE 'PICKED'
    END AS `Pick Status`

FROM document_items di

JOIN documents d
  ON d.id = di.document_id

JOIN products p
  ON p.id = di.product_id

LEFT JOIN units_of_measure u
  ON u.id = di.unit_id

LEFT JOIN v_inventory_balance ib
  ON ib.company_id = d.company_id
 AND ib.warehouse_id = d.warehouse_id
 AND ib.product_id = p.id

LEFT JOIN warehouse_location_products wlp
  ON wlp.company_id = d.company_id
 AND wlp.warehouse_id = d.warehouse_id
 AND wlp.product_id = p.id
 AND wlp.is_active = 1

LEFT JOIN warehouse_locations loc
  ON loc.id = wlp.location_id
 AND loc.is_active = 1

LEFT JOIN warehouse_pick_items wpi
  ON wpi.company_id = d.company_id
 AND wpi.document_item_id = di.id
 AND wpi.product_id = p.id

WHERE d.company_id = {{appsmith.store.companyId}}
  AND di.document_id = {{StockIssueDocumentsTable.selectedRow.documentId || StockIssueDocumentsTable.selectedRow["Issue ID"] || 0}}

GROUP BY
    di.id,
    d.id,
    d.document_number,
    d.warehouse_id,
    p.id,
    p.code,
    p.name,
    p.barcode,
    p.sku,
    u.code,
    di.quantity,
    ib.quantity_balance,
    loc.id,
    loc.code,
    loc.name

ORDER BY
    CASE
        WHEN COALESCE(SUM(wpi.picked_quantity), 0) >= di.quantity THEN 2
        ELSE 1
    END,
    p.name,
    loc.code;
