SELECT
    dp.id AS packageId,
    dp.package_number AS packageNumber,
    dp.package_barcode AS packageBarcode,
    dp.package_type AS packageType,
    dp.package_status AS packageStatus,

    dp.gross_weight AS grossWeight,
    dp.net_weight AS netWeight,
    dp.length_cm AS lengthCm,
    dp.width_cm AS widthCm,
    dp.height_cm AS heightCm,

    dp.item_count AS itemCount,
    dp.total_quantity AS totalQuantity,

    w.code AS warehouseCode,
    w.name AS warehouseName,

    sw.code AS sourceWarehouseCode,
    sw.name AS sourceWarehouseName,

    dw.code AS destinationWarehouseCode,
    dw.name AS destinationWarehouseName,

    bp.name AS partnerName,

    dp.note,
    dp.created_at AS createdAt

FROM delivery_packages dp
LEFT JOIN warehouses w ON w.id = dp.warehouse_id
LEFT JOIN warehouses sw ON sw.id = dp.source_warehouse_id
LEFT JOIN warehouses dw ON dw.id = dp.destination_warehouse_id
LEFT JOIN business_partners bp ON bp.id = dp.partner_id

WHERE (
    dp.document_id = {{ Number((this.params && this.params.documentId) || appsmith.store.inventorySelectedDocumentId || 0) }}
    OR dp.source_document_id = {{ Number((this.params && this.params.documentId) || appsmith.store.inventorySelectedDocumentId || 0) }}
)

ORDER BY dp.created_at DESC, dp.id DESC;
