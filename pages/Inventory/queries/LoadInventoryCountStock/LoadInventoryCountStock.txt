SELECT
    p.id AS productId,
    p.code AS productCode,
    p.name AS productName,
    p.barcode AS barcode,
    p.sku AS sku,
    p.unit_id AS unitId,
    u.code AS unitCode,

    COALESCE(ib.quantity_balance, 0) AS systemQuantity,
    0 AS countedQuantity,
    0 - COALESCE(ib.quantity_balance, 0) AS varianceQuantity,

    COALESCE(pp.price, 0) AS unitCost,
    (0 - COALESCE(ib.quantity_balance, 0)) * COALESCE(pp.price, 0) AS varianceValue,

    NULL AS warehouseLocationId,
    NULL AS locationCode,
    NULL AS note

FROM v_inventory_balance ib
JOIN products p ON p.id = ib.product_id
LEFT JOIN units_of_measure u ON u.id = p.unit_id
LEFT JOIN product_prices pp
  ON pp.product_id = p.id
 AND pp.price_type = 'PURCHASE'
 AND pp.is_active = 1

WHERE ib.company_id = {{appsmith.store.companyId}}
  AND ib.warehouse_id = {{this.params.warehouseId || NewCountWarehouseSelect.selectedOptionValue || 0}}
  AND COALESCE(ib.quantity_balance, 0) <> 0
  AND p.is_active = 1

ORDER BY p.code;
