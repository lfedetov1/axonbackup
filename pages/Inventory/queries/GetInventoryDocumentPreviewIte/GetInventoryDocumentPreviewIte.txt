SELECT
    di.id AS documentItemId,
    di.document_id AS documentId,
    di.line_no AS lineNo,

    p.id AS productId,
    p.code AS productCode,
    p.name AS productName,
    p.barcode AS barcode,

    COALESCE(di.description, p.name) AS description,

    u.code AS unitCode,

    COALESCE(di.quantity, 0) AS quantity,

    0 AS systemQuantity,
    0 AS countedQuantity,
    0 AS varianceQuantity,

    COALESCE(di.unit_price, 0) AS unitPrice,
    COALESCE(di.line_total, 0) AS lineTotal,

    NULL AS expiryDate,
    NULL AS batchNumber,

    di.note AS note

FROM document_items di
JOIN documents d ON d.id = di.document_id
LEFT JOIN products p ON p.id = di.product_id
LEFT JOIN units_of_measure u ON u.id = di.unit_id

WHERE d.id = {{ Number((this.params && this.params.documentId) || appsmith.store.inventorySelectedDocumentId || 0) }}
  AND d.company_id = {{ Number(appsmith.store.companyId || 1) }}
  AND d.document_type IN (
      'OPENING_STOCK',
      'INVENTORY_COUNT',
      'PURCHASE_RECEIPT',
      'INBOUND_DELIVERY',
      'STOCK_OUT',
      'STOCK_ISSUE',
      'TRANSFER_REQUEST',
      'SUPPLIER_RETURN',
      'GOODS_IN_TRANSIT',
      'DELIVERY_MANIFEST',
      'DISPATCH',
      'OUTBOUND_DELIVERY',
      'DELIVERY_DISPATCH',
      'PACKAGE_RECEIVING'
  )

ORDER BY di.line_no;
