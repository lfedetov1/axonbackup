SELECT
    sm.id AS movementId,
    sm.movement_date AS movementDate,
    sm.movement_type AS movementType,

    CASE
        WHEN sm.movement_type IN ('OUT', 'TRANSFER_OUT') THEN 'OUT'
        WHEN sm.movement_type IN ('IN', 'TRANSFER_IN') THEN 'IN'
        WHEN sm.movement_type = 'ADJUSTMENT' AND sm.quantity < 0 THEN 'OUT'
        WHEN sm.movement_type = 'ADJUSTMENT' THEN 'IN'
        ELSE sm.movement_type
    END AS movementDirection,

    w.id AS warehouseId,
    w.code AS warehouseCode,
    w.name AS warehouseName,

    p.id AS productId,
    p.code AS productCode,
    p.name AS productName,
    p.barcode AS barcode,
    p.sku AS sku,

    u.code AS unitCode,
    u.name AS unitName,

    sm.quantity AS quantity,

    CASE
        WHEN sm.movement_type IN ('OUT', 'TRANSFER_OUT') THEN -ABS(sm.quantity)
        WHEN sm.movement_type IN ('IN', 'TRANSFER_IN') THEN ABS(sm.quantity)
        WHEN sm.movement_type = 'ADJUSTMENT' THEN sm.quantity
        ELSE sm.quantity
    END AS signedQuantity,

    sm.unit_cost AS unitCost,
    sm.total_cost AS totalCost,

    d.id AS documentId,
    d.document_type AS documentType,
    d.document_number AS documentNumber,
    d.status AS documentStatus,
    d.document_date AS documentDate,

    sm.batch_number AS batchNumber,
    sm.serial_number AS serialNumber,
    sm.note AS note,
    sm.created_at AS createdAt

FROM stock_movements sm

JOIN warehouses w
  ON w.id = sm.warehouse_id

JOIN products p
  ON p.id = sm.product_id

LEFT JOIN units_of_measure u
  ON u.id = p.unit_id

LEFT JOIN documents d
  ON d.id = sm.document_id

WHERE sm.company_id = {{appsmith.store.companyId}}
  AND (
      {{InventoryWarehouseSelect.selectedOptionValue || appsmith.store.warehouseId || 0}} = 0
      OR sm.warehouse_id = {{InventoryWarehouseSelect.selectedOptionValue || appsmith.store.warehouseId || 0}}
  )
  AND (
    {{ StockMovementsSearchInput.text || "" }} = ""
    OR p.code LIKE CONCAT('%', {{ StockMovementsSearchInput.text || "" }}, '%')
    OR p.name LIKE CONCAT('%', {{ StockMovementsSearchInput.text || "" }}, '%')
    OR d.document_number LIKE CONCAT('%', {{ StockMovementsSearchInput.text || "" }}, '%')
)

AND sm.movement_date >= {{ moment(StockMovementsDateFrom.selectedDate || moment().startOf("month")).format("YYYY-MM-DD") }}
AND sm.movement_date < DATE_ADD({{ moment(StockMovementsDateTo.selectedDate || moment()).format("YYYY-MM-DD") }}, INTERVAL 1 DAY)

AND (
    {{ AppAccess.canViewAllWarehouses() ? 1 : 0 }} = 1
    OR sm.warehouse_id IN (
        {{ (appsmith.store.warehouseAccessIds || [0]).join(",") }}
    )
)

ORDER BY
    sm.movement_date DESC,
    sm.id DESC;
