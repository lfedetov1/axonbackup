SELECT
    d.id AS documentId,
    d.document_number AS issueNumber,
    d.document_date AS issueDate,
    d.status AS status,
    d.note AS note,

    w.code AS warehouseCode,
    w.name AS warehouseName,

    c.name AS companyName,
    c.legal_name AS companyLegalName,
    c.address_line1 AS companyAddressLine1,
    c.postal_code AS companyPostalCode,
    c.city AS companyCity,
    c.country_code AS companyCountryCode,
    c.tax_number AS companyTaxNumber,
    c.email AS companyEmail,
    c.phone AS companyPhone,
    c.logo_path AS companyLogoPath,

    {{appsmith.store.username || ''}} AS createdByUsername

FROM documents d

LEFT JOIN warehouses w
  ON w.id = d.warehouse_id

LEFT JOIN companies c
  ON c.id = d.company_id

WHERE d.company_id = {{appsmith.store.companyId}}
  AND d.id = {{this.params.documentId}}
LIMIT 1;
