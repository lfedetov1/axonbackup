SELECT
    p.id AS productId,
    p.code AS productCode,
    p.name AS productName,
    p.barcode AS barcode,
    p.sku AS sku,
    p.description AS description,
    p.sale_price AS salePrice,
    p.track_stock AS trackStock,
    p.is_active AS isActive,
    u.id AS unitId,
    u.code AS unitCode,
    u.name AS unitName,
    COALESCE(ib.quantity_balance, 0) AS availableStock
FROM products p
LEFT JOIN units_of_measure u
  ON u.id = p.unit_id
LEFT JOIN v_inventory_balance ib
  ON ib.company_id = p.company_id
 AND ib.product_id = p.id
 AND ib.warehouse_id = {{DeliveryNoteWarehouseSelectFor.selectedOptionValue || InventoryWarehouseSelect.selectedOptionValue || appsmith.store.warehouseId || 0}}
WHERE p.company_id = {{appsmith.store.companyId}}
  AND p.is_active = 1
  AND (
      p.code = TRIM({{this.params.lookup}})
      OR p.barcode = TRIM({{this.params.lookup}})
      OR p.sku = TRIM({{this.params.lookup}})
      OR p.name LIKE CONCAT('%', TRIM({{this.params.lookup}}), '%')
  )
ORDER BY
    CASE
      WHEN p.barcode = TRIM({{this.params.lookup}}) THEN 1
      WHEN p.code = TRIM({{this.params.lookup}}) THEN 2
      WHEN p.sku = TRIM({{this.params.lookup}}) THEN 3
      ELSE 4
    END,
    p.name
LIMIT 1;
