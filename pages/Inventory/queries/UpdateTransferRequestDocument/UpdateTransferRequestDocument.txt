SET
  document_date = {{moment(TransferRequestDateInput.selectedDate || moment()).format("YYYY-MM-DD")}},
  source_warehouse_id = {{TransferRequestSourceWarehouse.selectedOptionValue}},
  destination_warehouse_id = {{TransferRequestDestinationWare.selectedOptionValue}},
  subtotal_amount = {{this.params.totalAmount || 0}},
  total_amount = {{this.params.totalAmount || 0}},
  note = {{TransferRequestNoteInput.text || null}},
  updated_at = NOW()
WHERE id = {{this.params.documentId}}
  AND company_id = {{appsmith.store.companyId}}
  AND document_type = 'TRANSFER_REQUEST'
