SELECT
    d.id AS documentId,
    d.document_number AS documentNumber,
    d.document_date AS documentDate,
    d.created_at AS createdAt,
    DATE_FORMAT(d.created_at, '%d.%m.%Y %H:%i:%s') AS issueTime,
    d.status AS status,
    d.total_amount AS totalAmount,
    d.note AS note,

    src.document_number AS sourceDocumentNumber,

    bp.code AS supplierCode,
    bp.name AS supplierName,
    bp.tax_number AS supplierTaxNumber,
    bp.email AS supplierEmail,
    bp.phone AS supplierPhone,
    bp.address_line1 AS supplierAddressLine1,

    w.code AS warehouseCode,
    w.name AS warehouseName,

    c.name AS companyName,
    c.tax_number AS companyTaxNumber,
    c.email AS companyEmail,
    c.phone AS companyPhone,
    c.address_line1 AS companyAddressLine1,
    c.city AS companyCity,
    c.postal_code AS companyPostalCode,
    c.logo_path AS companyLogoPath,

    u.username AS createdBy
FROM documents d
LEFT JOIN documents src ON src.id = d.source_document_id
LEFT JOIN business_partners bp ON bp.id = d.partner_id
LEFT JOIN warehouses w ON w.id = d.warehouse_id
LEFT JOIN companies c ON c.id = d.company_id
LEFT JOIN users u ON u.id = d.created_by_user_id
WHERE d.id = {{this.params.documentId || SupplierReturnsTable.selectedRow?.documentId || 0}}
  AND d.company_id = {{appsmith.store.companyId}}
  AND d.document_type = 'PURCHASE_RETURN'
LIMIT 1;
