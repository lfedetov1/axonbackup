SELECT
    p.code AS productCode,
    p.name AS productName,
    COALESCE(SUM(di.quantity), 0) AS quantity,
    COALESCE(SUM(di.line_total), 0) AS totalAmount

FROM documents d

JOIN document_items di
  ON di.document_id = d.id

LEFT JOIN products p
  ON p.id = di.product_id

WHERE d.partner_id = {{ Number(appsmith.store.selectedPartnerId || 0) }}
  AND d.status <> 'CANCELLED'
  AND (
      {{ Number(appsmith.store.selectedPartnerCompanyId || appsmith.store.companyId || 0) }} = 0
      OR d.company_id = {{ Number(appsmith.store.selectedPartnerCompanyId || appsmith.store.companyId || 0) }}
  )

GROUP BY
    p.id,
    p.code,
    p.name

ORDER BY totalAmount DESC
LIMIT 10;
