SELECT
  u.id AS userId,
  u.username AS `Username`,
  u.email AS `Email`,
  u.first_name AS `First Name`,
  u.last_name AS `Last Name`,
  CONCAT(u.first_name, ' ', u.last_name) AS `Full Name`,
  u.phone AS `Phone`,
  CASE WHEN u.is_active = 1 THEN 'ACTIVE' ELSE 'INACTIVE' END AS `Status`,
  u.last_login_at AS `Last Login`,
  u.created_at AS `Created At`,
  u.updated_at AS `Updated At`,

  COALESCE(roles.roleCodes, '') AS `Roles`,
  COALESCE(warehouses.warehouseNames, '') AS `Warehouses`,

  CASE
    WHEN COALESCE(posAccess.id, 0) = 0 THEN 'NO'
    WHEN posAccess.is_active = 1 THEN 'YES'
    ELSE 'INACTIVE'
  END AS `POS Access`

FROM users u

LEFT JOIN (
  SELECT
    ura.user_id,
    GROUP_CONCAT(r.code ORDER BY r.code SEPARATOR ', ') AS roleCodes
  FROM user_role_assignments ura
  JOIN roles r ON r.id = ura.role_id
  GROUP BY ura.user_id
) roles ON roles.user_id = u.id

LEFT JOIN (
  SELECT
    uwa.user_id,
    GROUP_CONCAT(w.name ORDER BY w.name SEPARATOR ', ') AS warehouseNames
  FROM user_warehouse_access uwa
  JOIN warehouses w ON w.id = uwa.warehouse_id
  GROUP BY uwa.user_id
) warehouses ON warehouses.user_id = u.id

LEFT JOIN pos_user_access_codes posAccess
  ON posAccess.user_id = u.id
 AND posAccess.company_id = {{ Number(appsmith.store.companyId || 1) }}

WHERE
  (
    {{ AdminUserSearchInput.text || "" }} = ""
    OR u.username LIKE CONCAT('%', {{ AdminUserSearchInput.text || "" }}, '%')
    OR u.email LIKE CONCAT('%', {{ AdminUserSearchInput.text || "" }}, '%')
    OR u.first_name LIKE CONCAT('%', {{ AdminUserSearchInput.text || "" }}, '%')
    OR u.last_name LIKE CONCAT('%', {{ AdminUserSearchInput.text || "" }}, '%')
  )
  AND (
    {{ AdminUserStatusSelect.selectedOptionValue || "ALL" }} = "ALL"
    OR (
      {{ AdminUserStatusSelect.selectedOptionValue || "ALL" }} = "ACTIVE"
      AND u.is_active = 1
    )
    OR (
      {{ AdminUserStatusSelect.selectedOptionValue || "ALL" }} = "INACTIVE"
      AND u.is_active = 0
    )
  )

ORDER BY u.username;
