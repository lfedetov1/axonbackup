SELECT 'Sales Orders' AS stepLabel, 'SALES_ORDERS' AS actionCode, COUNT(*) AS countValue
FROM documents d
WHERE d.company_id = {{Number(appsmith.store.companyId || 0)}}
  AND d.document_type = 'SALES_ORDER'
  AND d.status NOT IN ('INVOICED', 'CANCELLED')

UNION ALL

SELECT 'Picking', 'WAREHOUSE_TASKS', COUNT(*)
FROM warehouse_tasks wt
WHERE wt.company_id = {{Number(appsmith.store.companyId || 0)}}
  AND wt.task_type = 'PICKING'
  AND wt.status IN ('OPEN', 'IN_PROGRESS', 'READY_FOR_NEXT_STEP')

UNION ALL

SELECT 'Packing', 'WAREHOUSE_TASKS', COUNT(*)
FROM warehouse_tasks wt
WHERE wt.company_id = {{Number(appsmith.store.companyId || 0)}}
  AND wt.task_type = 'PACKING'
  AND wt.status IN ('OPEN', 'IN_PROGRESS', 'READY_FOR_NEXT_STEP')

UNION ALL

SELECT 'Transfers', 'TRANSFERS', COUNT(*)
FROM documents d
WHERE d.company_id = {{Number(appsmith.store.companyId || 0)}}
  AND d.document_type = 'TRANSFER_REQUEST'
  AND d.status NOT IN ('RECEIVED', 'POSTED', 'CANCELLED')

UNION ALL

SELECT 'Fiscal', 'FISCAL_DOCS', COUNT(*)
FROM documents d
WHERE d.company_id = {{Number(appsmith.store.companyId || 0)}}
  AND d.document_type IN ('POS_SALE', 'SALES_INVOICE', 'CREDIT_NOTE')
  AND d.status <> 'CANCELLED'
  AND COALESCE(d.fiscal_status, '') NOT IN ('FISCALIZED', 'SUCCESS');
