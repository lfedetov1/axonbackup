SELECT
    d.document_date AS saleDate,
    SUM(COALESCE(di.quantity, 0)) AS soldQuantity,
    SUM(COALESCE(di.line_total, 0)) AS salesAmount,
    COUNT(DISTINCT d.id) AS documentCount

FROM documents d
JOIN document_items di ON di.document_id = d.id
JOIN products p ON p.id = di.product_id

WHERE d.company_id = {{ Number(appsmith.store.companyId || 1) }}
  AND d.document_type IN ('SALES_INVOICE', 'SALES_RECEIPT', 'POS_SALE')
  AND d.status <> 'CANCELLED'

  AND d.document_date >= {{
    appsmith.store.productDashboardDateFrom ||
    moment().startOf("month").format("YYYY-MM-DD")
  }}

  AND d.document_date < DATE_ADD(
    {{
      appsmith.store.productDashboardDateTo ||
      moment().format("YYYY-MM-DD")
    }},
    INTERVAL 1 DAY
  )

  AND (
    {{ Number(appsmith.store.productDashboardWarehouseId || 0) }} = 0
    OR d.warehouse_id =
       {{ Number(appsmith.store.productDashboardWarehouseId || 0) }}
  )

  AND (
    {{ AppAccess.canViewAllWarehouses() ? 1 : 0 }} = 1
    OR d.warehouse_id IN (
      {{ (appsmith.store.warehouseAccessIds || [0]).join(",") }}
    )
  )

  AND (
    {{ Number(appsmith.store.productDashboardCategoryId || 0) }} = 0
    OR p.category_id =
       {{ Number(appsmith.store.productDashboardCategoryId || 0) }}
  )

GROUP BY d.document_date
ORDER BY d.document_date;
