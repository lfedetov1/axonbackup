SELECT CONCAT(
    'PCR-',
    DATE_FORMAT(CURDATE(), '%Y%m'),
    '-',
    LPAD(
        COALESCE(
            MAX(CAST(SUBSTRING_INDEX(record_number, '-', -1) AS UNSIGNED)),
            0
        ) + 1,
        5,
        '0'
    )
) AS nextPriceChangeNumber
FROM product_price_change_records
WHERE company_id = {{ Number(appsmith.store.companyId || 1) }}
  AND YEAR(record_date) = YEAR(CURDATE())
  AND MONTH(record_date) = MONTH(CURDATE());
