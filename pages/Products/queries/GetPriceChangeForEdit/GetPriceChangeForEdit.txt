SELECT
    id AS priceChangeId,
    record_number AS recordNumber,
    record_date AS recordDate,
    effective_date AS effectiveDate,
    change_type AS changeType,
    status,
    reason,
    note
FROM product_price_change_records
WHERE id = {{ Number(this.params.priceChangeId || 0) }}
  AND company_id = {{ Number(appsmith.store.companyId || 1) }}
LIMIT 1;
