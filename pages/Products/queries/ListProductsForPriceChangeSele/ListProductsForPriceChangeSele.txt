SELECT
    p.id AS value,

    CONCAT(
        p.code,
        ' - ',
        p.name,
        CASE
            WHEN COALESCE(p.barcode, '') <> ''
            THEN CONCAT(' | ', p.barcode)
            ELSE ''
        END
    ) AS label,

    p.id AS productId,
    p.code AS productCode,
    p.name AS productName,
    COALESCE(p.barcode, '') AS barcode,
    COALESCE(p.sku, '') AS sku,

    COALESCE(current_price.price, 0) AS currentPrice,
    COALESCE(current_price.currency_code, 'EUR') AS currencyCode

FROM products p

LEFT JOIN product_prices current_price
    ON current_price.id = (
        SELECT pp.id
        FROM product_prices pp
        WHERE pp.product_id = p.id
          AND pp.price_type = 'SALE'
          AND pp.valid_from <= CURDATE()
          AND COALESCE(pp.valid_to, '9999-12-31') >= CURDATE()
        ORDER BY pp.valid_from DESC, pp.id DESC
        LIMIT 1
    )

WHERE p.company_id = {{ Number(appsmith.store.companyId || 1) }}

ORDER BY p.name, p.code;
