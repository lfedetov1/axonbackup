SELECT
    p.id AS productId,
    p.code AS productCode,
    p.name AS productName,
    p.barcode AS barcode,
    p.sku AS sku,

    COALESCE(u.code, 'kom') AS unitCode,
    COALESCE(current_price.currency_code, 'EUR') AS currencyCode,
    COALESCE(current_price.price, 0) AS salesPrice,

    COALESCE(
        (
            SELECT MIN(previous_price.price)
            FROM product_prices previous_price
            WHERE previous_price.product_id = p.id
              AND previous_price.price_type = 'SALE'
              AND previous_price.id <> current_price.id
              AND previous_price.valid_from < current_price.valid_from
              AND COALESCE(
                    previous_price.valid_to,
                    current_price.valid_from
                  ) >= DATE_SUB(current_price.valid_from, INTERVAL 30 DAY)
        ),
        current_price.price,
        0
    ) AS lowestPrice30Days,

    CASE
        WHEN current_price.price < COALESCE(
            (
                SELECT MIN(previous_price.price)
                FROM product_prices previous_price
                WHERE previous_price.product_id = p.id
                  AND previous_price.price_type = 'SALE'
                  AND previous_price.id <> current_price.id
                  AND previous_price.valid_from < current_price.valid_from
                  AND COALESCE(
                        previous_price.valid_to,
                        current_price.valid_from
                      ) >= DATE_SUB(current_price.valid_from, INTERVAL 30 DAY)
            ),
            current_price.price
        )
        THEN 1
        ELSE 0
    END AS isDiscounted,

    CASE
        WHEN current_price.price < COALESCE(
            (
                SELECT MIN(previous_price.price)
                FROM product_prices previous_price
                WHERE previous_price.product_id = p.id
                  AND previous_price.price_type = 'SALE'
                  AND previous_price.id <> current_price.id
                  AND previous_price.valid_from < current_price.valid_from
                  AND COALESCE(
                        previous_price.valid_to,
                        current_price.valid_from
                      ) >= DATE_SUB(current_price.valid_from, INTERVAL 30 DAY)
            ),
            current_price.price
        )
        THEN ROUND(
            (
                1 - current_price.price / COALESCE(
                    (
                        SELECT MIN(previous_price.price)
                        FROM product_prices previous_price
                        WHERE previous_price.product_id = p.id
                          AND previous_price.price_type = 'SALE'
                          AND previous_price.id <> current_price.id
                          AND previous_price.valid_from < current_price.valid_from
                          AND COALESCE(
                                previous_price.valid_to,
                                current_price.valid_from
                              ) >= DATE_SUB(current_price.valid_from, INTERVAL 30 DAY)
                    ),
                    current_price.price
                )
            ) * 100
        )
        ELSE 0
    END AS discountPercent

FROM products p

LEFT JOIN units_of_measure u
  ON u.id = p.unit_id

LEFT JOIN product_prices current_price
  ON current_price.id = (
      SELECT active_price.id
      FROM product_prices active_price
      WHERE active_price.product_id = p.id
        AND active_price.price_type = 'SALE'
        AND active_price.valid_from <= CURDATE()
        AND COALESCE(active_price.valid_to, '9999-12-31') >= CURDATE()
      ORDER BY active_price.valid_from DESC, active_price.id DESC
      LIMIT 1
  )

WHERE (
    {{ Number((this.params && this.params.productId) || 0) }} > 0
    AND p.id = {{ Number((this.params && this.params.productId) || 0) }}
)
OR (
    {{ Number((this.params && this.params.productId) || 0) }} = 0
    AND (
        p.barcode = {{ (this.params && this.params.lookup) || "" }}
        OR p.code = {{ (this.params && this.params.lookup) || "" }}
        OR p.sku = {{ (this.params && this.params.lookup) || "" }}
    )
)

LIMIT 1;
