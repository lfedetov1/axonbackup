SELECT
    d.id AS documentId,
    d.document_number AS `Receipt Number`,
    COALESCE(d.fiscal_number, d.document_number) AS `Fiscal Number`,

    d.document_date AS `Date`,
    DATE_FORMAT(d.created_at, '%d.%m.%Y %H:%i:%s') AS `Issue Time`,
    DATE_FORMAT(d.created_at, '%H:%i:%s') AS `Time`,

    COALESCE(bp.name, 'Walk-in Customer') AS `Customer`,

    d.payment_method AS `Payment`,
    d.card_type AS `Card`,
    d.currency_code AS `Currency`,

    COALESCE(d.subtotal_amount, 0) AS `Subtotal`,
    COALESCE(d.discount_amount, 0) AS `Discount`,
    COALESCE(d.tax_amount, 0) AS `Tax`,
    COALESCE(d.total_amount, 0) AS `Total`,
    COALESCE(d.paid_amount, 0) AS `Paid`,
    COALESCE(d.total_amount, 0) - COALESCE(d.paid_amount, 0) AS `Open Amount`,

    d.status AS `Status`,
    d.posting_status AS `Posting Status`,
    COALESCE(d.fiscal_status, '') AS `Fiscal Status`,

    w.code AS `Warehouse Code`,
    w.name AS `Warehouse`,

    u.username AS `Created By`,
    cu.username AS `Cashier`,

    d.created_at AS createdAt
FROM documents d
LEFT JOIN business_partners bp ON bp.id = d.partner_id
LEFT JOIN warehouses w ON w.id = d.warehouse_id
LEFT JOIN users u ON u.id = d.created_by_user_id
LEFT JOIN users cu ON cu.id = d.cashier_user_id
WHERE d.company_id = {{ Number(appsmith.store.companyId || 1) }}
  AND d.document_type = 'POS_SALE'

  AND (
      {{ POSReceiptSearchInput.text || "" }} = ""
      OR d.document_number LIKE CONCAT('%', {{ POSReceiptSearchInput.text || "" }}, '%')
      OR d.fiscal_number LIKE CONCAT('%', {{ POSReceiptSearchInput.text || "" }}, '%')
      OR bp.name LIKE CONCAT('%', {{ POSReceiptSearchInput.text || "" }}, '%')
      OR d.total_amount LIKE CONCAT('%', {{ POSReceiptSearchInput.text || "" }}, '%')
  )

  AND (
      {{ POSReceiptPaymentSelect.selectedOptionValue || "ALL" }} = "ALL"
      OR d.payment_method = {{ POSReceiptPaymentSelect.selectedOptionValue || "ALL" }}
  )

  AND (
      {{ POSReceiptStatusSelect.selectedOptionValue || "ALL" }} = "ALL"
      OR d.status = {{ POSReceiptStatusSelect.selectedOptionValue || "ALL" }}
  )
	AND (
    {{ Number(appsmith.store.selectedWarehouseId || appsmith.store.warehouseId1 || appsmith.store.warehouseId || 0) }} = 0
    OR d.warehouse_id = {{ Number(appsmith.store.selectedWarehouseId || appsmith.store.warehouseId1 || appsmith.store.warehouseId || 0) }}
)

  AND d.document_date >= {{ moment(POSReceiptDateFrom.selectedDate || moment()).format("YYYY-MM-DD") }}
  AND d.document_date <= {{ moment(POSReceiptDateTo.selectedDate || moment()).format("YYYY-MM-DD") }}

ORDER BY d.created_at DESC, d.id DESC
LIMIT 500;
