SELECT
    ct.id AS cashTransactionId,
    ct.reference_number AS referenceNumber,
    ct.transaction_type AS transactionType,
    ct.transaction_date AS transactionDate,
    ct.amount,
    CONCAT(
      ct.reference_number,
      ' | ',
      DATE_FORMAT(ct.transaction_date, '%d.%m.%Y %H:%i'),
      ' | ',
      FORMAT(ct.amount, 2),
      ' EUR'
    ) AS label,
    ct.reference_number AS value
FROM cash_transactions ct
WHERE ct.cash_register_id = {{PosReportCashRegisterSelect.selectedOptionValue || 0}}
  AND DATE(ct.transaction_date) >= {{PosReportDateFrom.selectedDate || moment().format("YYYY-MM-DD")}}
  AND DATE(ct.transaction_date) <= {{PosReportDateTo.selectedDate || moment().format("YYYY-MM-DD")}}
  AND (
    (
      {{PosReportTypeSelect.selectedOptionValue || ""}} = 'CASH_IN_SLIP'
      AND ct.transaction_type = 'IN'
    )
    OR (
      {{PosReportTypeSelect.selectedOptionValue || ""}} = 'CASH_OUT_SLIP'
      AND ct.transaction_type = 'OUT'
    )
    OR (
      {{PosReportTypeSelect.selectedOptionValue || ""}} = 'BANK_DEPOSIT_SLIP'
      AND ct.transaction_type = 'TRANSFER_OUT'
    )
    OR (
      {{PosReportTypeSelect.selectedOptionValue || ""}} = 'BANK_WITHDRAWAL_SLIP'
      AND ct.transaction_type = 'TRANSFER_IN'
    )
  )
ORDER BY ct.transaction_date DESC;
