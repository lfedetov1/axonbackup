SELECT
    id AS cashTransactionId,
    reference_number AS referenceNumber,
    transaction_date AS transactionDate,
    transaction_type AS transactionType,
    amount AS amount
FROM cash_transactions
WHERE cash_register_id = {{PosControlCashRegisterSelect.selectedOptionValue}}
  AND created_by_user_id = {{appsmith.store.userId}}
ORDER BY id DESC
LIMIT 1;
