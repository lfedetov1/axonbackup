SELECT
    ct.id AS transactionId,
    ct.transaction_date AS transactionDate,
    ct.transaction_type AS transactionType,
    ct.amount AS amount,
    ct.note AS note,
    cr.code AS cashRegisterCode,
    cr.name AS cashRegisterName
FROM cash_transactions ct
JOIN cash_registers cr
  ON cr.id = ct.cash_register_id
WHERE cr.company_id = {{appsmith.store.companyId}}
  AND ct.transaction_date >= {{moment(PosReportDateFrom.selectedDate || moment()).format("YYYY-MM-DD")}}
  AND ct.transaction_date < DATE_ADD({{moment(PosReportDateTo.selectedDate || moment()).format("YYYY-MM-DD")}}, INTERVAL 1 DAY)
  AND (
      {{Number(PosReportCashRegisterSelect.selectedOptionValue || 0)}} = 0
      OR ct.cash_register_id = {{Number(PosReportCashRegisterSelect.selectedOptionValue || 0)}}
  )
ORDER BY
    ct.transaction_date DESC
LIMIT 80;
